static int lengthOfLongestSubstring(String s) {
    Map<Character, Integer> last = new HashMap<>(); //@init
    int best = 0, left = 0; //@init2
    for (int right = 0; right < s.length(); right++) { //@loop
        char c = s.charAt(right);
        if (last.containsKey(c) && last.get(c) >= left) { //@check
            left = last.get(c) + 1; //@shrink
        }
        last.put(c, right); //@store
        if (right - left + 1 > best) { //@cmp
            best = right - left + 1; //@newBest
        }
    }
    return best; //@ret
}
