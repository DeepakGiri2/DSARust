package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
    "strings"
)

type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

func serialize(root *TreeNode) string {
    out := []string{}
    var dfs func(node *TreeNode)
    dfs = func(node *TreeNode) {
        if node == nil {
            out = append(out, "N")
            return
        }
        out = append(out, strconv.Itoa(node.Val))
        dfs(node.Left)
        dfs(node.Right)
    }
    dfs(root)
    return strings.Join(out, ",")
}

func deserialize(data string) *TreeNode {
    tokens := strings.Split(data, ",")
    idx := 0
    var dfs func() *TreeNode
    dfs = func() *TreeNode {
        tok := tokens[idx]
        idx++
        if tok == "N" {
            return nil
        }
        v, _ := strconv.Atoi(tok)
        node := &TreeNode{Val: v}
        node.Left = dfs()
        node.Right = dfs()
        return node
    }
    return dfs()
}

func buildTree(tokens []string) *TreeNode {
    if len(tokens) == 0 || tokens[0] == "n" {
        return nil
    }
    v, _ := strconv.Atoi(tokens[0])
    root := &TreeNode{Val: v}
    queue := []*TreeNode{root}
    i := 1
    for len(queue) > 0 && i < len(tokens) {
        cur := queue[0]
        queue = queue[1:]
        if i < len(tokens) {
            if tokens[i] != "n" {
                v, _ := strconv.Atoi(tokens[i])
                cur.Left = &TreeNode{Val: v}
                queue = append(queue, cur.Left)
            }
            i++
        }
        if i < len(tokens) {
            if tokens[i] != "n" {
                v, _ := strconv.Atoi(tokens[i])
                cur.Right = &TreeNode{Val: v}
                queue = append(queue, cur.Right)
            }
            i++
        }
    }
    return root
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    line, _ := reader.ReadString('\n')
    root := buildTree(strings.Fields(line))
    data := serialize(root)
    rebuilt := deserialize(data)
    fmt.Println(serialize(rebuilt))
}
