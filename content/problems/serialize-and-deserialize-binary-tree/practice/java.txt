import java.util.*;
import java.io.*;

class Main {

    static List<String> parts;
    static int idx;

    static String serialize(TreeNode root) {
        parts = new ArrayList<>();
        sdfs(root);
        return String.join(",", parts);
    }

    static void sdfs(TreeNode node) {
        if (node == null) {
            parts.add("N");
            return;
        }
        parts.add(String.valueOf(node.val));
        sdfs(node.left);
        sdfs(node.right);
    }

    static TreeNode deserialize(String data) {
        String[] tokens = data.split(",");
        idx = 0;
        return ddfs(tokens);
    }

    static TreeNode ddfs(String[] tokens) {
        String tok = tokens[idx];
        idx++;
        if (tok.equals("N")) {
            return null;
        }
        TreeNode node = new TreeNode(Integer.parseInt(tok));
        node.left = ddfs(tokens);
        node.right = ddfs(tokens);
        return node;
    }

    static class TreeNode {
        int val;
        TreeNode left, right;
        TreeNode(int v) { val = v; }
    }

    static TreeNode buildTree(String[] tokens) {
        if (tokens.length == 0 || tokens[0].equals("n")) return null;
        TreeNode root = new TreeNode(Integer.parseInt(tokens[0]));
        Queue<TreeNode> q = new ArrayDeque<>();
        q.add(root);
        int i = 1;
        while (!q.isEmpty() && i < tokens.length) {
            TreeNode cur = q.poll();
            if (i < tokens.length) {
                if (!tokens[i].equals("n")) { cur.left = new TreeNode(Integer.parseInt(tokens[i])); q.add(cur.left); }
                i++;
            }
            if (i < tokens.length) {
                if (!tokens[i].equals("n")) { cur.right = new TreeNode(Integer.parseInt(tokens[i])); q.add(cur.right); }
                i++;
            }
        }
        return root;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().trim().split("\\s+");
        TreeNode root = buildTree(tokens);
        String data = serialize(root);
        TreeNode rebuilt = deserialize(data);
        System.out.println(serialize(rebuilt));
    }
}
