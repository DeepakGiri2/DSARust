static int majorityElement(int[] nums) {
    int count = 0; //@init
    int candidate = 0; //@cand
    for (int i = 0; i < nums.length; i++) { //@loop
        if (count == 0) { //@zero
            candidate = nums[i]; //@pick
        }
        if (nums[i] == candidate) { //@same
            count++; //@up
        } else {
            count--; //@down
        }
    }
    return candidate; //@ret
}
