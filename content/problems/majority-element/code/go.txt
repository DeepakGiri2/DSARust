func majorityElement(nums []int) int {
    count := 0 //@init
    candidate := 0 //@cand
    for i := 0; i < len(nums); i++ { //@loop
        if count == 0 { //@zero
            candidate = nums[i] //@pick
        }
        if nums[i] == candidate { //@same
            count++ //@up
        } else {
            count-- //@down
        }
    }
    return candidate //@ret
}
