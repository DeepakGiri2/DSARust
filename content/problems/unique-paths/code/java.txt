static int uniquePaths(int m, int n) {
    int[][] dp = new int[m][n]; //@alloc
    for (int r = 0; r < m; r++) { //@outer
        for (int c = 0; c < n; c++) { //@inner
            if (r == 0 || c == 0) { //@edge
                dp[r][c] = 1; //@one
            } else {
                dp[r][c] = dp[r-1][c] + dp[r][c-1]; //@sum
            }
        }
    }
    return dp[m-1][n-1]; //@ret
}
