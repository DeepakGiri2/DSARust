static boolean canPartition(int[] nums) {
    int total = 0; //@sum
    for (int v : nums) {
        total += v;
    }
    if (total % 2 == 1) { //@odd
        return false;
    }
    int target = total / 2; //@target
    boolean[] reachable = new boolean[target + 1]; //@init
    reachable[0] = true; //@seed
    for (int v : nums) { //@loop
        for (int s = target; s >= v; s--) { //@inner
            if (reachable[s - v]) { //@check
                reachable[s] = true; //@mark
            }
        }
    }
    return reachable[target]; //@ret
}
