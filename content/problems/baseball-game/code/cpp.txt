int calPoints(vector<string>& operations) {
    vector<int> record; //@init
    for (auto& op : operations) { //@loop
        if (op == "C") { //@isC
            record.pop_back(); //@popC
        } else if (op == "D") { //@isD
            record.push_back(2 * record.back()); //@pushD
        } else if (op == "+") { //@isPlus
            int n = record.size();
            record.push_back(record[n - 1] + record[n - 2]); //@pushPlus
        } else { //@isNum
            record.push_back(stoi(op)); //@pushNum
        }
    }
    int total = 0; //@sum
    for (int v : record) { //@add
        total += v;
    }
    return total; //@ret
}
