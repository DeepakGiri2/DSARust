static ListNode swapPairs(ListNode head) {
    ListNode dummy = new ListNode(0, head); //@dummy
    ListNode prev = dummy; //@prev
    while (prev.next != null && prev.next.next != null) { //@loop
        ListNode first = prev.next; //@first
        ListNode second = first.next; //@second
        first.next = second.next; //@relinkFirst
        second.next = first; //@relinkSecond
        prev.next = second; //@relinkPrev
        prev = first; //@advance
    }
    return dummy.next; //@ret
}
