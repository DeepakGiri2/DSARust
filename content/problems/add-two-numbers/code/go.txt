func addTwoNumbers(l1 *ListNode, l2 *ListNode) *ListNode {
    dummy := &ListNode{} //@dummy
    tail, carry := dummy, 0 //@init
    for l1 != nil || l2 != nil || carry != 0 { //@loop
        sum := carry //@sum
        if l1 != nil { //@takeA
            sum += l1.Val
            l1 = l1.Next
        }
        if l2 != nil { //@takeB
            sum += l2.Val
            l2 = l2.Next
        }
        carry = sum / 10 //@carry
        tail.Next = &ListNode{Val: sum % 10} //@digit
        tail = tail.Next //@advance
    }
    return dummy.Next //@ret
}
