import java.util.*;
import java.io.*;

class Main {

    static List<Integer> topKFrequent(int[] nums, int k) {
        Map<Integer, Integer> count = new HashMap<>();
        for (int n : nums) {
            count.put(n, count.getOrDefault(n, 0) + 1);
        }
        List<List<Integer>> buckets = new ArrayList<>();
        for (int i = 0; i <= nums.length; i++) buckets.add(new ArrayList<>());
        Set<Integer> added = new HashSet<>();
        for (int n : nums) {
            if (!added.contains(n)) {
                buckets.get(count.get(n)).add(n);
                added.add(n);
            }
        }
        List<Integer> res = new ArrayList<>();
        for (int f = nums.length; f >= 1; f--) {
            for (int n : buckets.get(f)) {
                if (res.size() < k) {
                    res.add(n);
                }
            }
        }
        return res;
    }

    static int[] readInts(BufferedReader br) throws Exception {
        String line = br.readLine();
        if (line == null || line.trim().isEmpty()) return new int[0];
        String[] parts = line.trim().split("\\s+");
        int[] nums = new int[parts.length];
        for (int i = 0; i < parts.length; i++) nums[i] = Integer.parseInt(parts[i]);
        return nums;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] nums = readInts(br);
        int k = readInts(br)[0];
        List<Integer> res = topKFrequent(nums, k);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < res.size(); i++) {
            if (i > 0) sb.append(' ');
            sb.append(res.get(i));
        }
        System.out.println(sb.toString());
    }
}
