static List<Integer> topKFrequent(int[] nums, int k) {
    Map<Integer, Integer> count = new HashMap<>(); //@count
    for (int n : nums) { //@build
        count.put(n, count.getOrDefault(n, 0) + 1); //@inc
    }
    List<List<Integer>> buckets = new ArrayList<>(); //@buckets
    for (int i = 0; i <= nums.length; i++) buckets.add(new ArrayList<>());
    Set<Integer> added = new HashSet<>(); //@addedmap
    for (int n : nums) { //@fill
        if (!added.contains(n)) { //@isnew
            buckets.get(count.get(n)).add(n); //@place
            added.add(n); //@mark
        }
    }
    List<Integer> res = new ArrayList<>(); //@res
    for (int f = nums.length; f >= 1; f--) { //@scan
        for (int n : buckets.get(f)) { //@take
            if (res.size() < k) { //@room
                res.add(n); //@add
            }
        }
    }
    return res; //@return
}
