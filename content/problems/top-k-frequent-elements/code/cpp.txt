vector<int> topKFrequent(vector<int>& nums, int k) {
    unordered_map<int, int> count; //@count
    for (int n : nums) { //@build
        count[n]++; //@inc
    }
    vector<vector<int>> buckets(nums.size() + 1); //@buckets
    unordered_set<int> added; //@addedmap
    for (int n : nums) { //@fill
        if (!added.count(n)) { //@isnew
            buckets[count[n]].push_back(n); //@place
            added.insert(n); //@mark
        }
    }
    vector<int> res; //@res
    for (int f = (int)nums.size(); f >= 1; f--) { //@scan
        for (int n : buckets[f]) { //@take
            if ((int)res.size() < k) { //@room
                res.push_back(n); //@add
            }
        }
    }
    return res; //@return
}
