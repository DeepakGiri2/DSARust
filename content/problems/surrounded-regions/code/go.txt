func solve(board [][]byte) {
    rows, cols := len(board), len(board[0])
    for r := 0; r < rows; r++ { //@leftRight
        mark(board, r, 0)
        mark(board, r, cols-1)
    }
    for c := 0; c < cols; c++ { //@topBottom
        mark(board, 0, c)
        mark(board, rows-1, c)
    }
    for r := 0; r < rows; r++ { //@rowLoop
        for c := 0; c < cols; c++ { //@colLoop
            if board[r][c] == 'O' { //@captured
                board[r][c] = 'X' //@flip
            } else if board[r][c] == 'S' { //@safe
                board[r][c] = 'O' //@restore
            }
        }
    }
}

func mark(board [][]byte, r, c int) {
    if r < 0 || r >= len(board) || c < 0 || c >= len(board[0]) || board[r][c] != 'O' { //@bounds
        return
    }
    board[r][c] = 'S' //@save
    mark(board, r+1, c) //@down
    mark(board, r-1, c) //@up
    mark(board, r, c+1) //@right
    mark(board, r, c-1) //@left
}
