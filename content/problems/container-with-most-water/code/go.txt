func maxArea(heights []int) int {
    l, r := 0, len(heights)-1 //@init
    best := 0 //@best
    for l < r { //@loop
        h := heights[l] //@hmin
        if heights[r] < h { //@pick
            h = heights[r]
        }
        area := h * (r - l) //@area
        if area > best { //@cmp
            best = area //@update
        }
        if heights[l] < heights[r] { //@which
            l++ //@moveL
        } else {
            r-- //@moveR
        }
    }
    return best //@ret
}
