static int minSubArrayLen(int target, int[] nums) {
    int minLen = nums.length + 1; //@init
    int sum = 0, left = 0; //@init2
    for (int right = 0; right < nums.length; right++) { //@loop
        sum += nums[right]; //@add
        while (sum >= target) { //@shrink
            if (right - left + 1 < minLen) { //@cmp
                minLen = right - left + 1; //@newMin
            }
            sum -= nums[left]; //@sub
            left++; //@advance
        }
    }
    if (minLen == nums.length + 1) { //@test
        return 0; //@none
    }
    return minLen; //@ret
}
