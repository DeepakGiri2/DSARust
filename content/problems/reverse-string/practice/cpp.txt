#include <bits/stdc++.h>
using namespace std;

string reverseString(string s) {
    string chars = s;
    int l = 0, r = (int)chars.size() - 1;
    while (l < r) {
        swap(chars[l], chars[r]);
        l++;
        r--;
    }
    return chars;
}

int main() {
    string s;
    getline(cin, s);
    while (!s.empty() && (s.back() == '\n' || s.back() == '\r')) s.pop_back();
    cout << reverseString(s) << "\n";
}
