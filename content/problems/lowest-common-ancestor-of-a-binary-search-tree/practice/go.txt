package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
    "strings"
)

type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

func lowestCommonAncestor(root *TreeNode, p int, q int) int {
    cur := root
    for cur != nil {
        if p < cur.Val && q < cur.Val {
            cur = cur.Left
        } else if p > cur.Val && q > cur.Val {
            cur = cur.Right
        } else {
            return cur.Val
        }
    }
    return -1
}

func buildTree(tokens []string) *TreeNode {
    if len(tokens) == 0 || tokens[0] == "n" {
        return nil
    }
    v, _ := strconv.Atoi(tokens[0])
    root := &TreeNode{Val: v}
    queue := []*TreeNode{root}
    i := 1
    for len(queue) > 0 && i < len(tokens) {
        cur := queue[0]
        queue = queue[1:]
        if i < len(tokens) {
            if tokens[i] != "n" {
                v, _ := strconv.Atoi(tokens[i])
                cur.Left = &TreeNode{Val: v}
                queue = append(queue, cur.Left)
            }
            i++
        }
        if i < len(tokens) {
            if tokens[i] != "n" {
                v, _ := strconv.Atoi(tokens[i])
                cur.Right = &TreeNode{Val: v}
                queue = append(queue, cur.Right)
            }
            i++
        }
    }
    return root
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    l1, _ := reader.ReadString('\n')
    l2, _ := reader.ReadString('\n')
    root := buildTree(strings.Fields(l1))
    pq := strings.Fields(l2)
    p, _ := strconv.Atoi(pq[0])
    q, _ := strconv.Atoi(pq[1])
    fmt.Println(lowestCommonAncestor(root, p, q))
}
