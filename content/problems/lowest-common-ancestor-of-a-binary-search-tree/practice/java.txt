import java.util.*;
import java.io.*;

class Main {

    static int lowestCommonAncestor(TreeNode root, int p, int q) {
        TreeNode cur = root;
        while (cur != null) {
            if (p < cur.val && q < cur.val) {
                cur = cur.left;
            } else if (p > cur.val && q > cur.val) {
                cur = cur.right;
            } else {
                return cur.val;
            }
        }
        return -1;
    }

    static class TreeNode {
        int val;
        TreeNode left, right;
        TreeNode(int v) { val = v; }
    }

    static TreeNode buildTree(String[] tokens) {
        if (tokens.length == 0 || tokens[0].equals("n")) return null;
        TreeNode root = new TreeNode(Integer.parseInt(tokens[0]));
        Queue<TreeNode> q = new ArrayDeque<>();
        q.add(root);
        int i = 1;
        while (!q.isEmpty() && i < tokens.length) {
            TreeNode cur = q.poll();
            if (i < tokens.length) {
                if (!tokens[i].equals("n")) { cur.left = new TreeNode(Integer.parseInt(tokens[i])); q.add(cur.left); }
                i++;
            }
            if (i < tokens.length) {
                if (!tokens[i].equals("n")) { cur.right = new TreeNode(Integer.parseInt(tokens[i])); q.add(cur.right); }
                i++;
            }
        }
        return root;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tt = br.readLine().trim().split("\\s+");
        String[] pq = br.readLine().trim().split("\\s+");
        TreeNode root = buildTree(tt);
        int p = Integer.parseInt(pq[0]);
        int q = Integer.parseInt(pq[1]);
        System.out.println(lowestCommonAncestor(root, p, q));
    }
}
