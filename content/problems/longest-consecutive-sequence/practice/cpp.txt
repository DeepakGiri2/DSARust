#include <bits/stdc++.h>
using namespace std;

int longestConsecutive(vector<int>& nums) {
    unordered_set<int> set;
    for (int n : nums) {
        set.insert(n);
    }
    int best = 0;
    for (int n : set) {
        if (!set.count(n - 1)) {
            int length = 1;
            while (set.count(n + length)) {
                length++;
            }
            if (length > best) {
                best = length;
            }
        }
    }
    return best;
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> nums = readInts();
    cout << longestConsecutive(nums) << endl;
}
