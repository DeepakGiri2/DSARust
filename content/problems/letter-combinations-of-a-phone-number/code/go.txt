var keypad = map[byte]string{'2': "abc", '3': "def", '4': "ghi", '5': "jkl",
    '6': "mno", '7': "pqrs", '8': "tuv", '9': "wxyz"}

func letterCombinations(digits string) []string {
    if len(digits) == 0 { //@empty
        return []string{} //@retEmpty
    }
    res := []string{} //@init
    var back func(i int, cur string)
    back = func(i int, cur string) {
        if i == len(digits) { //@base
            res = append(res, cur) //@add
            return //@ret
        }
        for _, ch := range keypad[digits[i]] { //@loop
            back(i+1, cur+string(ch)) //@rec
        }
    }
    back(0, "") //@call
    return res //@done
}
