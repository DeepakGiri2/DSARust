int longestCommonSubsequence(string text1, string text2) {
    int m = text1.size(), n = text2.size(); //@init
    vector<vector<int>> dp(m+1, vector<int>(n+1, 0)); //@alloc
    for (int i = 1; i <= m; i++) { //@outer
        for (int j = 1; j <= n; j++) { //@inner
            if (text1[i-1] == text2[j-1]) { //@match
                dp[i][j] = dp[i-1][j-1] + 1; //@diag
            } else if (dp[i-1][j] >= dp[i][j-1]) { //@cmp
                dp[i][j] = dp[i-1][j]; //@up
            } else {
                dp[i][j] = dp[i][j-1]; //@left
            }
        }
    }
    return dp[m][n]; //@ret
}
