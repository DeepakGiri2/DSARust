static int longestCommonSubsequence(String text1, String text2) {
    int m = text1.length(), n = text2.length(); //@init
    int[][] dp = new int[m+1][n+1]; //@alloc
    for (int i = 1; i <= m; i++) { //@outer
        for (int j = 1; j <= n; j++) { //@inner
            if (text1.charAt(i-1) == text2.charAt(j-1)) { //@match
                dp[i][j] = dp[i-1][j-1] + 1; //@diag
            } else if (dp[i-1][j] >= dp[i][j-1]) { //@cmp
                dp[i][j] = dp[i-1][j]; //@up
            } else {
                dp[i][j] = dp[i][j-1]; //@left
            }
        }
    }
    return dp[m][n]; //@ret
}
