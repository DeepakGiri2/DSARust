#include <bits/stdc++.h>
using namespace std;

int longestCommonSubsequence(string text1, string text2) {
    int m = text1.size(), n = text2.size();
    vector<vector<int>> dp(m+1, vector<int>(n+1, 0));
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (text1[i-1] == text2[j-1]) {
                dp[i][j] = dp[i-1][j-1] + 1;
            } else if (dp[i-1][j] >= dp[i][j-1]) {
                dp[i][j] = dp[i-1][j];
            } else {
                dp[i][j] = dp[i][j-1];
            }
        }
    }
    return dp[m][n];
}

int main() {
    string text1, text2;
    getline(cin, text1);
    getline(cin, text2);
    while (!text1.empty() && (text1.back() == '\n' || text1.back() == '\r' || text1.back() == ' ')) text1.pop_back();
    while (!text2.empty() && (text2.back() == '\n' || text2.back() == '\r' || text2.back() == ' ')) text2.pop_back();
    cout << longestCommonSubsequence(text1, text2) << endl;
}
