#include <bits/stdc++.h>
using namespace std;

void dfs(vector<string>& grid, int r, int c);

int numIslands(vector<string>& grid) {
    int rows = grid.size(), cols = grid[0].size();
    int count = 0;
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == '1') {
                count++;
                dfs(grid, r, c);
            }
        }
    }
    return count;
}

void dfs(vector<string>& grid, int r, int c) {
    if (r < 0 || r >= (int)grid.size() || c < 0 || c >= (int)grid[0].size() || grid[r][c] != '1') {
        return;
    }
    grid[r][c] = '0';
    dfs(grid, r + 1, c);
    dfs(grid, r - 1, c);
    dfs(grid, r, c + 1);
    dfs(grid, r, c - 1);
}

int main() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<string> grid;
    string tk;
    while (ss >> tk) grid.push_back(tk);
    cout << numIslands(grid) << endl;
}
