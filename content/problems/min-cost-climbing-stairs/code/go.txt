func minCostClimbingStairs(cost []int) int {
    n := len(cost)
    dp := make([]int, n+1) //@init
    for i := n - 1; i >= 0; i-- { //@loop
        one := dp[i+1] //@one
        two := 0 //@twoInit
        if i+2 <= n { //@hasTwo
            two = dp[i+2] //@twoSet
        }
        dp[i] = cost[i] + min(one, two) //@set
    }
    return min(dp[0], dp[1]) //@ret
}
