int minCostClimbingStairs(vector<int>& cost) {
    int n = cost.size();
    vector<int> dp(n + 2, 0); //@init
    for (int i = n - 1; i >= 0; i--) { //@loop
        int one = dp[i + 1]; //@one
        int two = 0; //@twoInit
        if (i + 2 <= n) { //@hasTwo
            two = dp[i + 2]; //@twoSet
        }
        dp[i] = cost[i] + min(one, two); //@set
    }
    return min(dp[0], dp[1]); //@ret
}
