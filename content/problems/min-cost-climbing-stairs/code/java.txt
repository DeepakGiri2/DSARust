static int minCostClimbingStairs(int[] cost) {
    int n = cost.length;
    int[] dp = new int[n + 2]; //@init
    for (int i = n - 1; i >= 0; i--) { //@loop
        int one = dp[i + 1]; //@one
        int two = 0; //@twoInit
        if (i + 2 <= n) { //@hasTwo
            two = dp[i + 2]; //@twoSet
        }
        dp[i] = cost[i] + Math.min(one, two); //@set
    }
    return Math.min(dp[0], dp[1]); //@ret
}
