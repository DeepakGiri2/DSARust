int change(int amount, vector<int>& coins) {
    vector<int> dp(amount + 1, 0); //@init
    dp[0] = 1; //@seed
    for (int c : coins) { //@coinLoop
        for (int a = c; a <= amount; a++) { //@amountLoop
            int ways = dp[a - c]; //@lookup
            dp[a] += ways; //@add
        }
    }
    return dp[amount]; //@ret
}
