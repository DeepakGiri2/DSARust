static boolean isSubtree(TreeNode root, TreeNode subRoot) {
    if (root == null) { //@rootNil
        return false; //@retFalse
    }
    if (isSameTree(root, subRoot)) { //@check
        return true; //@retTrue
    }
    return isSubtree(root.left, subRoot) || isSubtree(root.right, subRoot); //@recurse
}

static boolean isSameTree(TreeNode p, TreeNode q) {
    if (p == null && q == null) { //@bothNil
        return true; //@sameTrue
    }
    if (p == null || q == null) { //@oneNil
        return false; //@sameFalse
    }
    if (p.val != q.val) { //@valDiff
        return false; //@sameDiff
    }
    return isSameTree(p.left, q.left) && isSameTree(p.right, q.right); //@sameRec
}
