func findItinerary(tickets [][]string) []string {
    adj := map[string][]string{} //@build
    for _, t := range tickets {
        adj[t[0]] = append(adj[t[0]], t[1])
    }
    for k := range adj { //@sort
        sort.Strings(adj[k])
    }
    route := []string{} //@route
    var visit func(string)
    visit = func(airport string) {
        for len(adj[airport]) > 0 { //@while
            next := adj[airport][0] //@take
            adj[airport] = adj[airport][1:]
            visit(next) //@recurse
        }
        route = append(route, airport) //@append
    }
    visit("JFK") //@start
    for i, j := 0, len(route)-1; i < j; i, j = i+1, j-1 { //@reverse
        route[i], route[j] = route[j], route[i]
    }
    return route //@ret
}
