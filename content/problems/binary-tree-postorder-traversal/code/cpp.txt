vector<int> postorderTraversal(TreeNode* root) {
    vector<int> out; //@init
    vector<TreeNode*> stack; //@stack
    if (root != nullptr) { //@seed
        stack.push_back(root);
    }
    while (!stack.empty()) { //@loop
        TreeNode* node = stack.back(); stack.pop_back(); //@pop
        out.push_back(node->val); //@visit
        if (node->left != nullptr) { //@hasL
            stack.push_back(node->left); //@pushL
        }
        if (node->right != nullptr) { //@hasR
            stack.push_back(node->right); //@pushR
        }
    }
    reverse(out.begin(), out.end()); //@reverse
    return out; //@ret
}
