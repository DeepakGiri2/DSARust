func postorderTraversal(root *TreeNode) []int {
    out := []int{} //@init
    stack := []*TreeNode{} //@stack
    if root != nil { //@seed
        stack = append(stack, root)
    }
    for len(stack) > 0 { //@loop
        node := stack[len(stack)-1] //@pop
        stack = stack[:len(stack)-1]
        out = append(out, node.Val) //@visit
        if node.Left != nil { //@hasL
            stack = append(stack, node.Left) //@pushL
        }
        if node.Right != nil { //@hasR
            stack = append(stack, node.Right) //@pushR
        }
    }
    for i, j := 0, len(out)-1; i < j; i, j = i+1, j-1 { //@reverse
        out[i], out[j] = out[j], out[i]
    }
    return out //@ret
}
