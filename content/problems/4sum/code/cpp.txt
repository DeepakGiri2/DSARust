vector<vector<int>> fourSum(vector<int>& nums, int target) {
    sort(nums.begin(), nums.end()); //@sort
    vector<vector<int>> res; //@init
    int n = nums.size();
    for (int i = 0; i + 3 < n; i++) { //@loopI
        if (i > 0 && nums[i] == nums[i - 1]) { //@dupI
            continue;
        }
        for (int j = i + 1; j + 2 < n; j++) { //@loopJ
            if (j > i + 1 && nums[j] == nums[j - 1]) { //@dupJ
                continue;
            }
            int lo = j + 1, hi = n - 1; //@pointers
            while (lo < hi) { //@squeeze
                long long sum = (long long)nums[i] + nums[j] + nums[lo] + nums[hi]; //@sum
                if (sum < target) { //@tooSmall
                    lo++;
                } else if (sum > target) { //@tooBig
                    hi--;
                } else { //@hit
                    res.push_back({nums[i], nums[j], nums[lo], nums[hi]}); //@record
                    lo++; //@skipLo
                    while (lo < hi && nums[lo] == nums[lo - 1]) lo++;
                    hi--; //@skipHi
                    while (lo < hi && nums[hi] == nums[hi + 1]) hi--;
                }
            }
        }
    }
    return res; //@ret
}
