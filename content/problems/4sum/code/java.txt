static List<List<Integer>> fourSum(int[] nums, int target) {
    Arrays.sort(nums); //@sort
    List<List<Integer>> res = new ArrayList<>(); //@init
    int n = nums.length;
    for (int i = 0; i + 3 < n; i++) { //@loopI
        if (i > 0 && nums[i] == nums[i - 1]) { //@dupI
            continue;
        }
        for (int j = i + 1; j + 2 < n; j++) { //@loopJ
            if (j > i + 1 && nums[j] == nums[j - 1]) { //@dupJ
                continue;
            }
            int lo = j + 1, hi = n - 1; //@pointers
            while (lo < hi) { //@squeeze
                long sum = (long) nums[i] + nums[j] + nums[lo] + nums[hi]; //@sum
                if (sum < target) { //@tooSmall
                    lo++;
                } else if (sum > target) { //@tooBig
                    hi--;
                } else { //@hit
                    res.add(List.of(nums[i], nums[j], nums[lo], nums[hi])); //@record
                    lo++; //@skipLo
                    while (lo < hi && nums[lo] == nums[lo - 1]) lo++;
                    hi--; //@skipHi
                    while (lo < hi && nums[hi] == nums[hi + 1]) hi--;
                }
            }
        }
    }
    return res; //@ret
}
