func multiply(num1 string, num2 string) string {
    if num1 == "0" || num2 == "0" { //@zero
        return "0"
    }
    m, n := len(num1), len(num2)
    res := make([]int, m+n) //@init
    for i := m - 1; i >= 0; i-- { //@outer
        for j := n - 1; j >= 0; j-- { //@inner
            p := int(num1[i]-'0') * int(num2[j]-'0') //@mul
            sum := p + res[i+j+1] //@place
            res[i+j+1] = sum % 10 //@digit
            res[i+j] += sum / 10 //@carry
        }
    }
    out := "" //@build
    for k, d := range res { //@trim
        if out == "" && d == 0 && k < len(res)-1 {
            continue
        }
        out += string(rune('0' + d))
    }
    return out //@ret
}
