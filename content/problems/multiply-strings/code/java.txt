static String multiply(String num1, String num2) {
    if (num1.equals("0") || num2.equals("0")) { //@zero
        return "0";
    }
    int m = num1.length(), n = num2.length();
    int[] res = new int[m + n]; //@init
    for (int i = m - 1; i >= 0; i--) { //@outer
        for (int j = n - 1; j >= 0; j--) { //@inner
            int p = (num1.charAt(i) - '0') * (num2.charAt(j) - '0'); //@mul
            int sum = p + res[i + j + 1]; //@place
            res[i + j + 1] = sum % 10; //@digit
            res[i + j] += sum / 10; //@carry
        }
    }
    StringBuilder out = new StringBuilder(); //@build
    for (int k = 0; k < res.length; k++) { //@trim
        if (out.length() == 0 && res[k] == 0 && k + 1 < res.length) {
            continue;
        }
        out.append((char) ('0' + res[k]));
    }
    return out.toString(); //@ret
}
