func subarraysDivByK(nums []int, k int) int {
    count := make([]int, k) //@init
    count[0] = 1 //@seed
    sum, ans := 0, 0 //@zero
    for _, x := range nums { //@loop
        sum += x //@add
        r := ((sum % k) + k) % k //@mod
        ans += count[r] //@take
        count[r]++ //@store
    }
    return ans //@ret
}
