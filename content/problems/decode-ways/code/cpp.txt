int numDecodings(string s) {
    if (s[0] == '0') { //@leadZero
        return 0;
    }
    int prev2 = 1, prev1 = 1; //@init
    for (int i = 1; i < (int)s.size(); i++) { //@loop
        int cur = 0; //@cur
        if (s[i] != '0') { //@single
            cur += prev1; //@addSingle
        }
        int two = (s[i - 1] - '0') * 10 + (s[i] - '0'); //@pair
        if (two >= 10 && two <= 26) { //@valid
            cur += prev2; //@addPair
        }
        prev2 = prev1; prev1 = cur; //@shift
    }
    return prev1; //@ret
}
