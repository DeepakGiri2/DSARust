func numDecodings(s string) int {
    if s[0] == '0' { //@leadZero
        return 0
    }
    prev2, prev1 := 1, 1 //@init
    for i := 1; i < len(s); i++ { //@loop
        cur := 0 //@cur
        if s[i] != '0' { //@single
            cur += prev1 //@addSingle
        }
        two := int(s[i-1]-'0')*10 + int(s[i]-'0') //@pair
        if two >= 10 && two <= 26 { //@valid
            cur += prev2 //@addPair
        }
        prev2, prev1 = prev1, cur //@shift
    }
    return prev1 //@ret
}
