pair<int, int> cellOf(int n, int square) {
    int idx = square - 1, row = idx / n, col = idx % n;
    if (row % 2 == 1) col = n - 1 - col;
    return {n - 1 - row, col};
}

int snakesAndLadders(vector<vector<int>>& board) {
    int n = board.size();
    int target = n * n; //@target
    vector<int> dist(n * n + 1, -1); //@init
    dist[1] = 0; //@seed
    queue<int> q; q.push(1); //@queue
    while (!q.empty()) { //@loop
        int cur = q.front(); q.pop(); //@pop
        if (cur == target) { //@arrived
            return dist[cur]; //@ret
        }
        for (int roll = 1; roll <= 6; roll++) { //@rolls
            int next = cur + roll; //@step
            if (next > target) { //@past
                break;
            }
            auto [r, c] = cellOf(n, next); //@locate
            if (board[r][c] != -1) { //@jump
                next = board[r][c]; //@teleport
            }
            if (dist[next] == -1) { //@fresh
                dist[next] = dist[cur] + 1; //@setDist
                q.push(next); //@push
            }
        }
    }
    return -1; //@none
}
