static int[] cellOf(int n, int square) {
    int idx = square - 1, row = idx / n, col = idx % n;
    if (row % 2 == 1) col = n - 1 - col;
    return new int[]{n - 1 - row, col};
}

static int snakesAndLadders(int[][] board) {
    int n = board.length;
    int target = n * n; //@target
    int[] dist = new int[n * n + 1]; //@init
    Arrays.fill(dist, -1);
    dist[1] = 0; //@seed
    Deque<Integer> q = new ArrayDeque<>(); q.addLast(1); //@queue
    while (!q.isEmpty()) { //@loop
        int cur = q.pollFirst(); //@pop
        if (cur == target) { //@arrived
            return dist[cur]; //@ret
        }
        for (int roll = 1; roll <= 6; roll++) { //@rolls
            int next = cur + roll; //@step
            if (next > target) { //@past
                break;
            }
            int[] rc = cellOf(n, next); //@locate
            if (board[rc[0]][rc[1]] != -1) { //@jump
                next = board[rc[0]][rc[1]]; //@teleport
            }
            if (dist[next] == -1) { //@fresh
                dist[next] = dist[cur] + 1; //@setDist
                q.addLast(next); //@push
            }
        }
    }
    return -1; //@none
}
