static int[] parent;

static int find(int x) {
    while (parent[x] != x) { //@walk
        parent[x] = parent[parent[x]];
        x = parent[x];
    }
    return x; //@root
}

static boolean validTree(int n, int[][] edges) {
    if (edges.length != n - 1) { //@count
        return false;
    }
    parent = new int[n]; //@init
    for (int i = 0; i < n; i++) {
        parent[i] = i; //@seed
    }
    for (int[] e : edges) { //@loop
        int a = find(e[0]), b = find(e[1]); //@find
        if (a == b) { //@cycle
            return false; //@no
        }
        parent[a] = b; //@union
    }
    return true; //@yes
}
