static List<List<String>> res = new ArrayList<>(); //@init
static List<String> cur = new ArrayList<>(); //@cur

static boolean isPalindrome(String s, int lo, int hi) {
    while (lo < hi) {
        if (s.charAt(lo) != s.charAt(hi)) return false;
        lo++; hi--;
    }
    return true;
}

static void back(String s, int start) {
    if (start == s.length()) { //@complete
        res.add(new ArrayList<>(cur)); //@record
        return; //@retDone
    }
    for (int end = start; end < s.length(); end++) { //@loop
        if (!isPalindrome(s, start, end)) { //@check
            continue;
        }
        cur.add(s.substring(start, end + 1)); //@choose
        back(s, end + 1); //@recurse
        cur.remove(cur.size() - 1); //@undo
    }
}

static List<List<String>> partition(String s) {
    res.clear(); cur.clear();
    back(s, 0); //@call
    return res; //@ret
}
