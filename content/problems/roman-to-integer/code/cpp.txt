int romanToInt(string s) {
    unordered_map<char, int> val = {{'I',1},{'V',5},{'X',10},{'L',50},{'C',100},{'D',500},{'M',1000}}; //@table
    int total = 0; //@init
    for (int i = 0; i < (int)s.size(); i++) { //@loop
        int v = val[s[i]]; //@lookup
        if (i + 1 < (int)s.size() && v < val[s[i + 1]]) { //@smaller
            total -= v; //@subtract
        } else { //@bigger
            total += v; //@add
        }
    }
    return total; //@ret
}
