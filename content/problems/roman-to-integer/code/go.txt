func romanToInt(s string) int {
    val := map[byte]int{'I': 1, 'V': 5, 'X': 10, 'L': 50, 'C': 100, 'D': 500, 'M': 1000} //@table
    total := 0 //@init
    for i := 0; i < len(s); i++ { //@loop
        v := val[s[i]] //@lookup
        if i+1 < len(s) && v < val[s[i+1]] { //@smaller
            total -= v //@subtract
        } else { //@bigger
            total += v //@add
        }
    }
    return total //@ret
}
