static int romanToInt(String s) {
    Map<Character, Integer> val = Map.of('I',1,'V',5,'X',10,'L',50,'C',100,'D',500,'M',1000); //@table
    int total = 0; //@init
    for (int i = 0; i < s.length(); i++) { //@loop
        int v = val.get(s.charAt(i)); //@lookup
        if (i + 1 < s.length() && v < val.get(s.charAt(i + 1))) { //@smaller
            total -= v; //@subtract
        } else { //@bigger
            total += v; //@add
        }
    }
    return total; //@ret
}
