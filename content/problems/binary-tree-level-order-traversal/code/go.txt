func levelOrder(root *TreeNode) [][]int {
    res := [][]int{} //@init
    if root == nil { //@empty
        return res //@retempty
    }
    queue := []*TreeNode{root} //@enq
    for len(queue) > 0 { //@while
        size := len(queue) //@size
        level := []int{} //@newlevel
        for i := 0; i < size; i++ { //@front
            node := queue[0] //@pop
            queue = queue[1:]
            level = append(level, node.Val) //@visit
            if node.Left != nil { //@left
                queue = append(queue, node.Left)
            }
            if node.Right != nil { //@right
                queue = append(queue, node.Right)
            }
        }
        res = append(res, level) //@pushlevel
    }
    return res //@ret
}
