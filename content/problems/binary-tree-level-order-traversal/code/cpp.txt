vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> res; //@init
    if (root == nullptr) { //@empty
        return res; //@retempty
    }
    queue<TreeNode*> q; //@enq
    q.push(root);
    while (!q.empty()) { //@while
        int size = q.size(); //@size
        vector<int> level; //@newlevel
        for (int i = 0; i < size; i++) { //@front
            TreeNode* node = q.front(); q.pop(); //@pop
            level.push_back(node->val); //@visit
            if (node->left != nullptr) { //@left
                q.push(node->left);
            }
            if (node->right != nullptr) { //@right
                q.push(node->right);
            }
        }
        res.push_back(level); //@pushlevel
    }
    return res; //@ret
}
