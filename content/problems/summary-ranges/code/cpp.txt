vector<string> summaryRanges(vector<int>& nums) {
    vector<string> out; //@init
    for (int i = 0; i < (int)nums.size(); ) { //@loop
        int start = i; //@start
        while (i + 1 < (int)nums.size() && nums[i + 1] == nums[i] + 1) { //@extend
            i++; //@grow
        }
        if (start == i) { //@single
            out.push_back(to_string(nums[start])); //@one
        } else {
            out.push_back(to_string(nums[start]) + "->" + to_string(nums[i])); //@range
        }
        i++; //@next
    }
    return out; //@ret
}
