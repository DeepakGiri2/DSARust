static int minimumTotal(List<List<Integer>> triangle) {
    int n = triangle.size();
    int[] dp = new int[n];
    for (int c = 0; c < n; c++) dp[c] = triangle.get(n - 1).get(c); //@init
    for (int r = n - 2; r >= 0; r--) { //@row
        for (int c = 0; c <= r; c++) { //@col
            dp[c] = triangle.get(r).get(c) + Math.min(dp[c], dp[c + 1]); //@relax
        }
    }
    return dp[0]; //@ret
}
