static boolean checkValidString(String s) {
    int lo = 0, hi = 0; //@init
    for (int i = 0; i < s.length(); i++) { //@loop
        char c = s.charAt(i);
        if (c == '(') { //@open
            lo++;
            hi++;
        } else if (c == ')') { //@close
            lo--;
            hi--;
        } else { //@star
            lo--;
            hi++;
        }
        if (hi < 0) { //@impossible
            return false; //@no
        }
        if (lo < 0) { //@clamp
            lo = 0;
        }
    }
    return lo == 0; //@ret
}
