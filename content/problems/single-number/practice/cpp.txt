#include <bits/stdc++.h>
using namespace std;

int singleNumber(vector<int>& nums) {
    int acc = 0;
    for (int i = 0; i < (int)nums.size(); i++) {
        acc ^= nums[i];
    }
    return acc;
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> nums = readInts();
    cout << singleNumber(nums) << endl;
}
