static TreeNode searchBST(TreeNode root, int val) {
    TreeNode cur = root; //@cur
    while (cur != null) { //@loop
        if (val == cur.val) { //@hit
            return cur; //@ret
        }
        if (val < cur.val) { //@less
            cur = cur.left; //@goL
        } else { //@greater
            cur = cur.right; //@goR
        }
    }
    return null; //@miss
}
