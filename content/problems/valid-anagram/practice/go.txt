package main

import (
    "bufio"
    "fmt"
    "os"
    "strings"
)

func isAnagram(s string, t string) bool {
    if len(s) != len(t) {
        return false
    }
    count := map[rune]int{}
    for _, c := range s {
        count[c]++
    }
    for _, c := range t {
        count[c]--
        if count[c] < 0 {
            return false
        }
    }
    return true
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    line1, _ := reader.ReadString('\n')
    line2, _ := reader.ReadString('\n')
    s := strings.TrimSpace(line1)
    t := strings.TrimSpace(line2)
    fmt.Println(isAnagram(s, t))
}
