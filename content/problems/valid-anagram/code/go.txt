func isAnagram(s string, t string) bool {
    if len(s) != len(t) { //@len
        return false //@lenbad
    }
    count := map[rune]int{} //@init
    for _, c := range s { //@builds
        count[c]++ //@inc
    }
    for _, c := range t { //@scant
        count[c]-- //@dec
        if count[c] < 0 { //@check
            return false //@bad
        }
    }
    return true //@ok
}
