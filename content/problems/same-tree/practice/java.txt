import java.util.*;
import java.io.*;

class Main {

    static boolean isSameTree(TreeNode p, TreeNode q) {
        if (p == null && q == null) {
            return true;
        }
        if (p == null || q == null) {
            return false;
        }
        if (p.val != q.val) {
            return false;
        }
        boolean left = isSameTree(p.left, q.left);
        boolean right = isSameTree(p.right, q.right);
        return left && right;
    }

    static class TreeNode {
        int val;
        TreeNode left, right;
        TreeNode(int v) { val = v; }
    }

    static TreeNode buildTree(String[] tokens) {
        if (tokens.length == 0 || tokens[0].equals("n")) return null;
        TreeNode root = new TreeNode(Integer.parseInt(tokens[0]));
        Queue<TreeNode> q = new ArrayDeque<>();
        q.add(root);
        int i = 1;
        while (!q.isEmpty() && i < tokens.length) {
            TreeNode cur = q.poll();
            if (i < tokens.length) {
                if (!tokens[i].equals("n")) { cur.left = new TreeNode(Integer.parseInt(tokens[i])); q.add(cur.left); }
                i++;
            }
            if (i < tokens.length) {
                if (!tokens[i].equals("n")) { cur.right = new TreeNode(Integer.parseInt(tokens[i])); q.add(cur.right); }
                i++;
            }
        }
        return root;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tp = br.readLine().trim().split("\\s+");
        String[] tq = br.readLine().trim().split("\\s+");
        TreeNode p = buildTree(tp);
        TreeNode q = buildTree(tq);
        System.out.println(isSameTree(p, q));
    }
}
