bool exist(vector<string>& board, string word) {
    int rows = board.size(), cols = board[0].size();
    for (int r = 0; r < rows; r++) { //@rowloop
        for (int c = 0; c < cols; c++) { //@colloop
            if (dfs(board, word, r, c, 0)) { //@calldfs
                return true; //@rettrue
            }
        }
    }
    return false; //@retfalse
}

bool dfs(vector<string>& board, string& word, int r, int c, int k) {
    if (k == (int)word.size()) { //@found
        return true; //@retok
    }
    if (r < 0 || r >= (int)board.size() || c < 0 || c >= (int)board[0].size() || board[r][c] != word[k]) { //@bounds
        return false; //@retbad
    }
    board[r][c] = '#'; //@mark
    bool found = dfs(board, word, r + 1, c, k + 1) || //@down
        dfs(board, word, r - 1, c, k + 1) || //@up
        dfs(board, word, r, c + 1, k + 1) || //@right
        dfs(board, word, r, c - 1, k + 1); //@left
    board[r][c] = word[k]; //@restore
    return found; //@retback
}
