#include <bits/stdc++.h>
using namespace std;

bool dfs(vector<string>& board, string& word, int r, int c, int k);

bool exist(vector<string>& board, string word) {
    int rows = board.size(), cols = board[0].size();
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (dfs(board, word, r, c, 0)) {
                return true;
            }
        }
    }
    return false;
}

bool dfs(vector<string>& board, string& word, int r, int c, int k) {
    if (k == (int)word.size()) {
        return true;
    }
    if (r < 0 || r >= (int)board.size() || c < 0 || c >= (int)board[0].size() || board[r][c] != word[k]) {
        return false;
    }
    board[r][c] = '#';
    bool found = dfs(board, word, r + 1, c, k + 1) ||
        dfs(board, word, r - 1, c, k + 1) ||
        dfs(board, word, r, c + 1, k + 1) ||
        dfs(board, word, r, c - 1, k + 1);
    board[r][c] = word[k];
    return found;
}

int main() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<string> board;
    string tk;
    while (ss >> tk) board.push_back(tk);
    string word;
    getline(cin, word);
    while (!word.empty() && isspace((unsigned char)word.back())) word.pop_back();
    cout << (exist(board, word) ? "true" : "false") << endl;
}
