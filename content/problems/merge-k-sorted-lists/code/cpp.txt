ListNode* mergeKLists(vector<ListNode*>& lists) {
    if (lists.empty()) { //@empty
        return nullptr; //@retNil
    }
    while (lists.size() > 1) { //@round
        vector<ListNode*> merged; //@newRound
        for (size_t i = 0; i < lists.size(); i += 2) { //@pairs
            ListNode* b = (i + 1 < lists.size()) ? lists[i + 1] : nullptr; //@hasPair
            merged.push_back(mergeTwo(lists[i], b)); //@merge
        }
        lists = merged; //@replace
    }
    return lists[0]; //@ret
}
