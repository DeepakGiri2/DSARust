int reverse(int x) {
    int sign = 1; //@sign
    long v = x;
    if (v < 0) {
        sign = -1;
        v = -v;
    }
    long out = 0; //@init
    while (v > 0) { //@loop
        int d = v % 10; //@digit
        if (out > (2147483647L - d) / 10) { //@overflow
            return 0; //@fail
        }
        out = out * 10 + d; //@push
        v /= 10; //@drop
    }
    return (int)(sign * out); //@ret
}
