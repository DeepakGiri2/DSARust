class KthLargest {
    int k;
    priority_queue<int, vector<int>, greater<int>> h;
public:
    KthLargest(int k, vector<int>& nums) { //@ctor
        this->k = k; //@init
        for (int n : nums) { //@ctorLoop
            add(n);
        }
    }

    int add(int val) { //@add
        h.push(val); //@push
        if ((int)h.size() > k) { //@check
            h.pop(); //@pop
        }
        return h.top(); //@return
    }
};
