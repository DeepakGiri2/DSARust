func setZeroes(matrix [][]int) {
    rows, cols := len(matrix), len(matrix[0])
    firstRow := false //@flag
    for c := 0; c < cols; c++ { //@scanFirst
        if matrix[0][c] == 0 {
            firstRow = true
        }
    }
    for r := 1; r < rows; r++ { //@markRows
        for c := 0; c < cols; c++ { //@markCols
            if matrix[r][c] == 0 { //@found
                matrix[0][c] = 0 //@markCol
                matrix[r][0] = 0 //@markRow
            }
        }
    }
    for r := 1; r < rows; r++ { //@applyRows
        for c := 1; c < cols; c++ { //@applyCols
            if matrix[0][c] == 0 || matrix[r][0] == 0 { //@check
                matrix[r][c] = 0 //@zero
            }
        }
    }
    if matrix[0][0] == 0 { //@colZero
        for r := 0; r < rows; r++ { //@wipeCol
            matrix[r][0] = 0
        }
    }
    if firstRow { //@rowZero
        for c := 0; c < cols; c++ { //@wipeRow
            matrix[0][c] = 0
        }
    }
}
