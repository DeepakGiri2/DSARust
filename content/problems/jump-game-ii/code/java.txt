static int jump(int[] nums) {
    int jumps = 0, end = 0, far = 0; //@init
    for (int i = 0; i < nums.length - 1; i++) { //@loop
        if (i + nums[i] > far) { //@reach
            far = i + nums[i]; //@setFar
        }
        if (i == end) { //@boundary
            jumps++; //@count
            end = far; //@extend
        }
    }
    return jumps; //@ret
}
