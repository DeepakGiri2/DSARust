func guessNumber(n int) int {
    lo, hi := 1, n //@bounds
    for lo <= hi { //@loop
        mid := lo + (hi-lo)/2 //@mid
        r := guess(mid) //@ask
        if r == 0 { //@hit
            return mid //@ret
        }
        if r < 0 { //@lower
            hi = mid - 1 //@shrinkHi
        } else { //@higher
            lo = mid + 1 //@shrinkLo
        }
    }
    return -1 //@miss
}
