string predictPartyVictory(string senate) {
    queue<int> radiant, dire; //@init
    int n = senate.size(); //@n
    for (int i = 0; i < n; i++) { //@fill
        if (senate[i] == 'R') radiant.push(i);
        else dire.push(i);
    }
    while (!radiant.empty() && !dire.empty()) { //@loop
        int r = radiant.front(); radiant.pop(); //@popR
        int d = dire.front(); dire.pop(); //@popD
        if (r < d) { //@earlier
            radiant.push(r + n); //@requeueR
        } else { //@later
            dire.push(d + n); //@requeueD
        }
    }
    if (!radiant.empty()) { //@check
        return "Radiant"; //@retR
    }
    return "Dire"; //@retD
}
