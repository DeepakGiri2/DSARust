bool isRange(string& s, int i, int j) {
    while (i < j) { //@rloop
        if (s[i] != s[j]) { //@rcmp
            return false; //@rbad
        }
        i++; //@rmoveL
        j--; //@rmoveR
    }
    return true; //@rgood
}

bool validPalindrome(string s) {
    int l = 0, r = (int)s.size() - 1; //@init
    while (l < r) { //@loop
        if (s[l] != s[r]) { //@cmp
            return isRange(s, l + 1, r) || isRange(s, l, r - 1); //@branch
        }
        l++; //@moveL
        r--; //@moveR
    }
    return true; //@good
}
