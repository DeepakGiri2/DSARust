import java.util.*;
import java.io.*;

class Main {

    static List<List<Integer>> combinationSum(int[] candidates, int target) {
        List<List<Integer>> res = new ArrayList<>();
        List<Integer> cur = new ArrayList<>();
        backtrack(0, target, candidates, cur, res);
        return res;
    }

    static void backtrack(int i, int remaining, int[] candidates, List<Integer> cur, List<List<Integer>> res) {
        if (remaining == 0) {
            res.add(new ArrayList<>(cur));
            return;
        }
        if (remaining < 0 || i == candidates.length) {
            return;
        }
        cur.add(candidates[i]);
        backtrack(i, remaining - candidates[i], candidates, cur, res);
        cur.remove(cur.size() - 1);
        backtrack(i + 1, remaining, candidates, cur, res);
    }

    static int[] readInts(BufferedReader br) throws Exception {
        String line = br.readLine();
        if (line == null || line.trim().isEmpty()) return new int[0];
        String[] parts = line.trim().split("\\s+");
        int[] nums = new int[parts.length];
        for (int i = 0; i < parts.length; i++) nums[i] = Integer.parseInt(parts[i]);
        return nums;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] candidates = readInts(br);
        int target = Integer.parseInt(br.readLine().trim());
        List<List<Integer>> res = combinationSum(candidates, target);
        StringBuilder sb = new StringBuilder();
        for (List<Integer> s : res) {
            for (int i = 0; i < s.size(); i++) {
                if (i > 0) sb.append(' ');
                sb.append(s.get(i));
            }
            sb.append('\n');
        }
        System.out.print(sb);
    }
}
