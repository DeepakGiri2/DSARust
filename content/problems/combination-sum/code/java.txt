static List<List<Integer>> combinationSum(int[] candidates, int target) {
    List<List<Integer>> res = new ArrayList<>(); //@init
    List<Integer> cur = new ArrayList<>(); //@cur
    backtrack(0, target, candidates, cur, res); //@call
    return res; //@done
}

static void backtrack(int i, int remaining, int[] candidates, List<Integer> cur, List<List<Integer>> res) {
    if (remaining == 0) { //@base
        res.add(new ArrayList<>(cur)); //@add
        return; //@ret
    }
    if (remaining < 0 || i == candidates.length) { //@prune
        return; //@pret
    }
    cur.add(candidates[i]); //@include
    backtrack(i, remaining - candidates[i], candidates, cur, res); //@recIn
    cur.remove(cur.size() - 1); //@exclude
    backtrack(i + 1, remaining, candidates, cur, res); //@recEx
}
