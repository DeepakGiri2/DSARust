func shipWithinDays(weights []int, days int) int {
    lo, hi := 0, 0 //@bounds
    for _, w := range weights { //@span
        if w > lo {
            lo = w
        }
        hi += w
    }
    for lo < hi { //@loop
        mid := lo + (hi-lo)/2 //@mid
        if feasible(weights, days, mid) { //@check
            hi = mid //@smaller
        } else { //@fail
            lo = mid + 1 //@larger
        }
    }
    return lo //@ret
}

func feasible(weights []int, days, capacity int) bool {
    need, load := 1, 0 //@start
    for _, w := range weights { //@scan
        if load+w > capacity { //@overflow
            need++ //@newDay
            load = 0
        }
        load += w //@add
    }
    return need <= days //@verdict
}
