package main

import (
    "bufio"
    "fmt"
    "os"
    "strings"
)

func checkInclusion(s1 string, s2 string) bool {
    n := len(s1)
    if n > len(s2) {
        return false
    }
    need := [26]int{}
    win := [26]int{}
    for i := 0; i < n; i++ {
        need[s1[i]-'a']++
        win[s2[i]-'a']++
    }
    for right := n - 1; right < len(s2); right++ {
        if right >= n {
            win[s2[right]-'a']++
            win[s2[right-n]-'a']--
        }
        ok := true
        for c := 0; c < 26; c++ {
            if win[c] != need[c] {
                ok = false
                break
            }
        }
        if ok {
            return true
        }
    }
    return false
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    line1, _ := reader.ReadString('\n')
    s1 := strings.TrimSpace(line1)
    line2, _ := reader.ReadString('\n')
    s2 := strings.TrimSpace(line2)
    fmt.Println(checkInclusion(s1, s2))
}
