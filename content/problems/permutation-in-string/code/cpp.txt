bool checkInclusion(string s1, string s2) {
    int n = (int)s1.size(); //@n
    if (n > (int)s2.size()) { //@guard
        return false; //@guardRet
    }
    vector<int> need(26, 0); //@need
    vector<int> win(26, 0); //@win
    for (int i = 0; i < n; i++) { //@build
        need[s1[i] - 'a']++; //@bn
        win[s2[i] - 'a']++; //@bw
    }
    for (int right = n - 1; right < (int)s2.size(); right++) { //@loop
        if (right >= n) { //@slide
            win[s2[right] - 'a']++; //@addR
            win[s2[right - n] - 'a']--; //@subL
        }
        bool ok = true; //@ok
        for (int c = 0; c < 26; c++) { //@compare
            if (win[c] != need[c]) { //@diff
                ok = false; //@setno
                break; //@brk
            }
        }
        if (ok) { //@match
            return true; //@found
        }
    }
    return false; //@ret
}
