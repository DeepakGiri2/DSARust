static boolean checkInclusion(String s1, String s2) {
    int n = s1.length(); //@n
    if (n > s2.length()) { //@guard
        return false; //@guardRet
    }
    int[] need = new int[26]; //@need
    int[] win = new int[26]; //@win
    for (int i = 0; i < n; i++) { //@build
        need[s1.charAt(i) - 'a']++; //@bn
        win[s2.charAt(i) - 'a']++; //@bw
    }
    for (int right = n - 1; right < s2.length(); right++) { //@loop
        if (right >= n) { //@slide
            win[s2.charAt(right) - 'a']++; //@addR
            win[s2.charAt(right - n) - 'a']--; //@subL
        }
        boolean ok = true; //@ok
        for (int c = 0; c < 26; c++) { //@compare
            if (win[c] != need[c]) { //@diff
                ok = false; //@setno
                break; //@brk
            }
        }
        if (ok) { //@match
            return true; //@found
        }
    }
    return false; //@ret
}
