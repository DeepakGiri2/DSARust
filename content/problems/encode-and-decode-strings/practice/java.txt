import java.util.*;
import java.io.*;

class Main {

    static String encode(List<String> strs) {
        StringBuilder encoded = new StringBuilder();
        for (String s : strs) {
            encoded.append(s.length()).append('#').append(s);
        }
        return encoded.toString();
    }

    static List<String> decode(String s) {
        List<String> res = new ArrayList<>();
        int i = 0;
        while (i < s.length()) {
            int j = i;
            while (s.charAt(j) != '#') j++;
            int length = Integer.parseInt(s.substring(i, j));
            String word = s.substring(j + 1, j + 1 + length);
            res.add(word);
            i = j + 1 + length;
        }
        return res;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null) line = "";
        List<String> words = new ArrayList<>();
        for (String w : line.trim().split("\\s+")) if (!w.isEmpty()) words.add(w);
        String encoded = encode(words);
        List<String> decoded = decode(encoded);
        System.out.println(String.join(" ", decoded));
    }
}
