static String longestPalindrome(String s) {
    String best = ""; //@init
    for (int i = 0; i < s.length(); i++) { //@loop
        String odd = expand(s, i, i); //@odd
        if (odd.length() > best.length()) { //@cmpOdd
            best = odd; //@setOdd
        }
        String even = expand(s, i, i + 1); //@even
        if (even.length() > best.length()) { //@cmpEven
            best = even; //@setEven
        }
    }
    return best; //@ret
}
