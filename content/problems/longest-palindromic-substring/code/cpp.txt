string longestPalindrome(string s) {
    string best = ""; //@init
    for (int i = 0; i < (int)s.size(); i++) { //@loop
        string odd = expand(s, i, i); //@odd
        if (odd.size() > best.size()) { //@cmpOdd
            best = odd; //@setOdd
        }
        string even = expand(s, i, i + 1); //@even
        if (even.size() > best.size()) { //@cmpEven
            best = even; //@setEven
        }
    }
    return best; //@ret
}
