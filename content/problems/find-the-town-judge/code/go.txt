func findJudge(n int, trust [][]int) int {
    score := make([]int, n+1) //@init
    for _, t := range trust { //@loop
        score[t[0]]-- //@truster
        score[t[1]]++ //@trusted
    }
    for i := 1; i <= n; i++ { //@scan
        if score[i] == n-1 { //@check
            return i //@ret
        }
    }
    return -1 //@none
}
