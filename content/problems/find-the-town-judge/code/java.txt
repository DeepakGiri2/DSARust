static int findJudge(int n, int[][] trust) {
    int[] score = new int[n + 1]; //@init
    for (int[] t : trust) { //@loop
        score[t[0]]--; //@truster
        score[t[1]]++; //@trusted
    }
    for (int i = 1; i <= n; i++) { //@scan
        if (score[i] == n - 1) { //@check
            return i; //@ret
        }
    }
    return -1; //@none
}
