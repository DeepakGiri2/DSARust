static int findMinArrowShots(int[][] points) {
    Arrays.sort(points, (a, b) -> Integer.compare(a[1], b[1])); //@sort
    int arrows = 1; //@init
    int end = points[0][1]; //@first
    for (int i = 1; i < points.length; i++) { //@loop
        if (points[i][0] > end) { //@miss
            arrows++; //@shoot
            end = points[i][1]; //@newend
        }
    }
    return arrows; //@ret
}
