vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
    if (!root) { //@empty
        return {}; //@noroot
    }
    vector<vector<int>> res; //@init
    queue<TreeNode*> q;
    q.push(root); //@seed
    bool leftToRight = true; //@flag
    while (!q.empty()) { //@while
        int size = q.size(); //@size
        vector<int> level(size); //@level
        for (int i = 0; i < size; i++) { //@pop
            TreeNode* node = q.front();
            q.pop();
            int pos = i; //@pos
            if (!leftToRight) { //@flip
                pos = size - 1 - i;
            }
            level[pos] = node->val; //@place
            if (node->left) { //@left
                q.push(node->left);
            }
            if (node->right) { //@right
                q.push(node->right);
            }
        }
        res.push_back(level); //@row
        leftToRight = !leftToRight; //@toggle
    }
    return res; //@ret
}
