static List<List<Integer>> zigzagLevelOrder(TreeNode root) {
    if (root == null) { //@empty
        return new ArrayList<>(); //@noroot
    }
    List<List<Integer>> res = new ArrayList<>(); //@init
    Deque<TreeNode> q = new ArrayDeque<>();
    q.add(root); //@seed
    boolean leftToRight = true; //@flag
    while (!q.isEmpty()) { //@while
        int size = q.size(); //@size
        Integer[] level = new Integer[size]; //@level
        for (int i = 0; i < size; i++) { //@pop
            TreeNode node = q.poll();
            int pos = i; //@pos
            if (!leftToRight) { //@flip
                pos = size - 1 - i;
            }
            level[pos] = node.val; //@place
            if (node.left != null) { //@left
                q.add(node.left);
            }
            if (node.right != null) { //@right
                q.add(node.right);
            }
        }
        res.add(Arrays.asList(level)); //@row
        leftToRight = !leftToRight; //@toggle
    }
    return res; //@ret
}
