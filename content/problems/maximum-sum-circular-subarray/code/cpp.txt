int maxSubarraySumCircular(vector<int>& nums) {
    int total = 0;
    int maxSum = nums[0], curMax = 0;
    int minSum = nums[0], curMin = 0; //@init
    for (int x : nums) { //@loop
        curMax = max(curMax + x, x); //@grow
        maxSum = max(maxSum, curMax); //@best
        curMin = min(curMin + x, x); //@shrink
        minSum = min(minSum, curMin); //@worst
        total += x; //@total
    }
    if (maxSum > 0) { //@check
        return max(maxSum, total - minSum); //@wrap
    }
    return maxSum; //@allneg
}
