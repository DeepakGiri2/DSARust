static boolean canJump(int[] nums) {
    int reach = 0; //@init
    for (int i = 0; i < nums.length; i++) { //@loop
        if (i > reach) { //@check
            return false; //@bad
        }
        if (i + nums[i] > reach) { //@cmp
            reach = i + nums[i]; //@update
        }
    }
    return true; //@ret
}
