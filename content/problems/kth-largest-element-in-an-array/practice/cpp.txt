#include <bits/stdc++.h>
using namespace std;

int findKthLargest(vector<int>& nums, int k) {
    priority_queue<int, vector<int>, greater<int>> h;
    for (int n : nums) {
        h.push(n);
        if ((int)h.size() > k) {
            h.pop();
        }
    }
    return h.top();
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> nums = readInts();
    int k = readInts()[0];
    cout << findKthLargest(nums, k) << endl;
}
