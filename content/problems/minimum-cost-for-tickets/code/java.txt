static int mincostTickets(int[] days, int[] costs) {
    int last = days[days.length - 1]; //@span
    boolean[] travel = new boolean[last + 1]; //@mark
    for (int d : days) { //@markLoop
        travel[d] = true;
    }
    int[] dp = new int[last + 1]; //@init
    for (int d = 1; d <= last; d++) { //@loop
        if (!travel[d]) { //@rest
            dp[d] = dp[d - 1]; //@carry
            continue;
        }
        int one = dp[Math.max(0, d - 1)] + costs[0]; //@one
        int seven = dp[Math.max(0, d - 7)] + costs[1]; //@seven
        int thirty = dp[Math.max(0, d - 30)] + costs[2]; //@thirty
        dp[d] = Math.min(one, Math.min(seven, thirty)); //@best
    }
    return dp[last]; //@ret
}
