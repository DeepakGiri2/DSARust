static int[] asteroidCollision(int[] asteroids) {
    Deque<Integer> stack = new ArrayDeque<>(); //@init
    for (int a : asteroids) { //@loop
        boolean alive = true; //@alive
        while (alive && a < 0 && !stack.isEmpty() && stack.peek() > 0) { //@collide
            int top = stack.peek(); //@peek
            if (top < -a) { //@smaller
                stack.pop(); //@popTop
            } else if (top == -a) { //@equal
                stack.pop(); //@popBoth
                alive = false;
            } else { //@bigger
                alive = false; //@die
            }
        }
        if (alive) { //@survives
            stack.push(a); //@push
        }
    }
    int[] out = new int[stack.size()];
    for (int i = out.length - 1; i >= 0; i--) out[i] = stack.pop();
    return out; //@ret
}
