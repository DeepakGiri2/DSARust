class DetectSquares {
    map<pair<int, int>, int> cnt; //@init
    vector<pair<int, int>> pts; //@list
public:
    void add(vector<int> p) {
        cnt[{p[0], p[1]}]++; //@add
        pts.push_back({p[0], p[1]}); //@remember
    }
    int count(vector<int> p) {
        int x = p[0], y = p[1]; //@query
        int total = 0; //@total
        for (auto& q : pts) { //@scan
            int px = q.first, py = q.second;
            if (abs(px - x) != abs(py - y) || px == x || py == y) { //@diagonal
                continue;
            }
            total += cnt[{px, y}] * cnt[{x, py}]; //@corners
        }
        return total; //@ret
    }
};
