vector<vector<int>> permute(vector<int>& nums) {
    vector<vector<int>> res; //@init
    vector<int> current; //@cur
    vector<bool> used(nums.size(), false); //@used
    backtrack(nums, current, used, res); //@call
    return res; //@done
}

void backtrack(vector<int>& nums, vector<int>& current, vector<bool>& used, vector<vector<int>>& res) {
    if (current.size() == nums.size()) { //@base
        res.push_back(current); //@add
        return; //@ret
    }
    for (int i = 0; i < (int)nums.size(); i++) { //@loop
        if (used[i]) { //@skip
            continue;
        }
        used[i] = true; //@mark
        current.push_back(nums[i]); //@choose
        backtrack(nums, current, used, res); //@rec
        current.pop_back(); //@unchoose
        used[i] = false; //@unmark
    }
}
