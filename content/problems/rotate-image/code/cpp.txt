void rotate(vector<vector<int>>& matrix) {
    int n = matrix.size();
    for (int i = 0; i < n; i++) { //@tOuter
        for (int j = i + 1; j < n; j++) { //@tInner
            swap(matrix[i][j], matrix[j][i]); //@swap
        }
    }
    for (int i = 0; i < n; i++) { //@rOuter
        for (int l = 0, r = n - 1; l < r; l++, r--) { //@rInner
            swap(matrix[i][l], matrix[i][r]); //@rSwap
        }
    }
}
