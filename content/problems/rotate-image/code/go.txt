func rotate(matrix [][]int) {
    n := len(matrix)
    for i := 0; i < n; i++ { //@tOuter
        for j := i + 1; j < n; j++ { //@tInner
            matrix[i][j], matrix[j][i] = matrix[j][i], matrix[i][j] //@swap
        }
    }
    for i := 0; i < n; i++ { //@rOuter
        for l, r := 0, n-1; l < r; l, r = l+1, r-1 { //@rInner
            matrix[i][l], matrix[i][r] = matrix[i][r], matrix[i][l] //@rSwap
        }
    }
}
