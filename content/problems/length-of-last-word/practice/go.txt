package main

import (
    "bufio"
    "fmt"
    "os"
    "strings"
)

func lengthOfLastWord(s string) int {
    i := len(s) - 1
    length := 0
    for i >= 0 && s[i] == ' ' {
        i--
    }
    for i >= 0 && s[i] != ' ' {
        length++
        i--
    }
    return length
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    line, _ := reader.ReadString('\n')
    line = strings.TrimRight(line, "\r\n")
    fmt.Println(lengthOfLastWord(line))
}
