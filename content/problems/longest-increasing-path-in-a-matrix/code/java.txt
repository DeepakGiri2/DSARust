static int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

static int dfs(int[][] m, int[][] memo, int r, int c) {
    if (memo[r][c] != 0) { //@cached
        return memo[r][c]; //@retCached
    }
    int best = 1; //@start
    for (int[] d : dirs) { //@neighbours
        int nr = r + d[0], nc = c + d[1];
        if (nr >= 0 && nr < m.length && nc >= 0 && nc < m[0].length &&
            m[nr][nc] > m[r][c]) { //@higher
            int l = 1 + dfs(m, memo, nr, nc); //@recurse
            if (l > best) { //@better
                best = l; //@take
            }
        }
    }
    memo[r][c] = best; //@memo
    return best; //@retBest
}

static int longestIncreasingPath(int[][] matrix) {
    int rows = matrix.length, cols = matrix[0].length;
    int[][] memo = new int[rows][cols]; //@init
    int best = 0; //@best
    for (int r = 0; r < rows; r++) { //@rowLoop
        for (int c = 0; c < cols; c++) { //@colLoop
            int l = dfs(matrix, memo, r, c); //@call
            if (l > best) { //@cmp
                best = l; //@update
            }
        }
    }
    return best; //@ret
}
