int lengthOfLIS(vector<int>& nums) {
    int n = nums.size(); //@init
    vector<int> dp(n); //@dp
    int best = 0; //@best0
    for (int i = 0; i < n; i++) { //@outer
        dp[i] = 1; //@seed
        for (int j = 0; j < i; j++) { //@inner
            if (nums[j] < nums[i] && dp[j]+1 > dp[i]) { //@check
                dp[i] = dp[j] + 1; //@extend
            }
        }
        if (dp[i] > best) { //@cmp
            best = dp[i]; //@newbest
        }
    }
    return best; //@ret
}
