#include <bits/stdc++.h>
using namespace std;

int removeElement(vector<int>& nums, int val) {
    int k = 0;
    for (int i = 0; i < (int)nums.size(); i++) {
        if (nums[i] != val) {
            nums[k] = nums[i];
            k++;
        }
    }
    return k;
}

int main() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    int val;
    cin >> val;
    int k = removeElement(nums, val);
    cout << k << "\n";
    for (int i = 0; i < k; i++) {
        if (i) cout << " ";
        cout << nums[i];
    }
    cout << endl;
}
