func isInterleave(s1 string, s2 string, s3 string) bool {
    m, n := len(s1), len(s2)
    if m+n != len(s3) { //@length
        return false
    }
    dp := make([][]bool, m+1) //@init
    for i := range dp {
        dp[i] = make([]bool, n+1)
    }
    dp[0][0] = true //@seed
    for i := 0; i <= m; i++ { //@rowLoop
        for j := 0; j <= n; j++ { //@colLoop
            if i == 0 && j == 0 { //@origin
                continue
            }
            ok := false //@ok
            if i > 0 && s1[i-1] == s3[i+j-1] && dp[i-1][j] { //@fromS1
                ok = true //@takeS1
            }
            if j > 0 && s2[j-1] == s3[i+j-1] && dp[i][j-1] { //@fromS2
                ok = true //@takeS2
            }
            dp[i][j] = ok //@store
        }
    }
    return dp[m][n] //@ret
}
