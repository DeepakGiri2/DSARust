func longestOnes(nums []int, k int) int {
    left, zeros, best := 0, 0, 0 //@init
    for right := 0; right < len(nums); right++ { //@grow
        if nums[right] == 0 { //@iszero
            zeros++ //@count
        }
        for zeros > k { //@shrink
            if nums[left] == 0 { //@leftzero
                zeros-- //@give
            }
            left++ //@move
        }
        if right-left+1 > best { //@best
            best = right - left + 1 //@set
        }
    }
    return best //@ret
}
