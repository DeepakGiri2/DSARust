int maxCoins(vector<int>& nums) {
    int n = nums.size();
    vector<int> vals(n + 2, 1); //@pad
    for (int i = 0; i < n; i++) vals[i + 1] = nums[i];
    vector<vector<int>> dp(n + 2, vector<int>(n + 2, 0)); //@init
    for (int length = 1; length <= n; length++) { //@len
        for (int left = 1; left + length - 1 <= n; left++) { //@left
            int right = left + length - 1; //@right
            for (int k = left; k <= right; k++) { //@last
                int gain = vals[left - 1] * vals[k] * vals[right + 1]; //@gain
                int total = dp[left][k - 1] + gain + dp[k + 1][right]; //@total
                if (total > dp[left][right]) { //@better
                    dp[left][right] = total; //@store
                }
            }
        }
    }
    return dp[1][n]; //@ret
}
