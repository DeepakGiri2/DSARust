bool isMatch(string s, string p) {
    int m = s.size(), n = p.size();
    vector<vector<bool>> dp(m + 1, vector<bool>(n + 1, false)); //@init
    dp[0][0] = true; //@seed
    for (int j = 1; j <= n; j++) { //@emptyRow
        if (p[j - 1] == '*') { //@starEmpty
            dp[0][j] = dp[0][j - 2]; //@copyEmpty
        }
    }
    for (int i = 1; i <= m; i++) { //@rowLoop
        for (int j = 1; j <= n; j++) { //@colLoop
            if (p[j - 1] == '*') { //@isStar
                dp[i][j] = dp[i][j - 2]; //@zeroUse
                if (p[j - 2] == '.' || p[j - 2] == s[i - 1]) { //@canRepeat
                    dp[i][j] = dp[i][j] || dp[i - 1][j]; //@moreUse
                }
            } else if (p[j - 1] == '.' || p[j - 1] == s[i - 1]) { //@single
                dp[i][j] = dp[i - 1][j - 1]; //@shift
            }
        }
    }
    return dp[m][n]; //@ret
}
