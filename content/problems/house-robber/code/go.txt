func rob(nums []int) int {
    n := len(nums) //@init
    dp := make([]int, n) //@dp
    dp[0] = nums[0] //@base
    for i := 1; i < n; i++ { //@loop
        skip := dp[i-1] //@skip
        rob := nums[i] //@robval
        if i >= 2 { //@guard
            rob += dp[i-2] //@add
        }
        if rob > skip { //@cmp
            dp[i] = rob //@takerob
        } else {
            dp[i] = skip //@takeskip
        }
    }
    return dp[n-1] //@ret
}
