static TreeNode sortedArrayToBST(int[] nums) {
    return build(nums, 0, nums.length - 1); //@call
}

static TreeNode build(int[] nums, int lo, int hi) {
    if (lo > hi) { //@base
        return null;
    }
    int mid = lo + (hi - lo) / 2; //@mid
    TreeNode node = new TreeNode(nums[mid]); //@make
    node.left = build(nums, lo, mid - 1); //@recL
    node.right = build(nums, mid + 1, hi); //@recR
    return node; //@ret
}
