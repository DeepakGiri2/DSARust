static void merge(int[] nums1, int m, int[] nums2, int n) {
    int i = m - 1, j = n - 1, k = m + n - 1; //@init
    while (j >= 0) { //@loop
        if (i >= 0 && nums1[i] > nums2[j]) { //@cmp
            nums1[k] = nums1[i]; //@take1
            i--; //@decI
        } else {
            nums1[k] = nums2[j]; //@take2
            j--; //@decJ
        }
        k--; //@decK
    }
}
