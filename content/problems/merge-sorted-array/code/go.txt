func merge(nums1 []int, m int, nums2 []int, n int) {
    i, j, k := m-1, n-1, m+n-1 //@init
    for j >= 0 { //@loop
        if i >= 0 && nums1[i] > nums2[j] { //@cmp
            nums1[k] = nums1[i] //@take1
            i-- //@decI
        } else {
            nums1[k] = nums2[j] //@take2
            j-- //@decJ
        }
        k-- //@decK
    }
}
