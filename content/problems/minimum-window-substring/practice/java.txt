import java.util.*;
import java.io.*;

class Main {

    static String minWindow(String s, String t) {
        if (t.length() == 0) {
            return "";
        }
        Map<Character, Integer> need = new HashMap<>();
        for (int i = 0; i < t.length(); i++) {
            need.merge(t.charAt(i), 1, Integer::sum);
        }
        Map<Character, Integer> window = new HashMap<>();
        int have = 0, needCount = need.size();
        int resL = -1, resR = -1, resLen = Integer.MAX_VALUE;
        int left = 0;
        for (int right = 0; right < s.length(); right++) {
            char c = s.charAt(right);
            window.merge(c, 1, Integer::sum);
            if (need.containsKey(c) && window.get(c).equals(need.get(c))) {
                have++;
            }
            while (have == needCount) {
                if (right - left + 1 < resLen) {
                    resL = left; resR = right;
                    resLen = right - left + 1;
                }
                window.merge(s.charAt(left), -1, Integer::sum);
                if (need.containsKey(s.charAt(left)) && window.get(s.charAt(left)) < need.get(s.charAt(left))) {
                    have--;
                }
                left++;
            }
        }
        if (resLen == Integer.MAX_VALUE) {
            return "";
        }
        return s.substring(resL, resR + 1);
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine().trim();
        String t = br.readLine().trim();
        System.out.println(minWindow(s, t));
    }
}
