#include <bits/stdc++.h>
using namespace std;

int maxProfit(vector<int>& prices) {
    int minPrice = prices[0], best = 0;
    for (int day = 1; day < (int)prices.size(); day++) {
        int profit = prices[day] - minPrice;
        if (profit > best) {
            best = profit;
        }
        if (prices[day] < minPrice) {
            minPrice = prices[day];
        }
    }
    return best;
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> prices = readInts();
    cout << maxProfit(prices) << endl;
}
