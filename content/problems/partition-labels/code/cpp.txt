vector<int> partitionLabels(string s) {
    unordered_map<char, int> last; //@last
    for (int i = 0; i < (int)s.size(); i++) { //@scan
        last[s[i]] = i; //@record
    }
    vector<int> res; //@init
    int start = 0, end = 0; //@bounds
    for (int i = 0; i < (int)s.size(); i++) { //@loop
        if (last[s[i]] > end) { //@extend
            end = last[s[i]]; //@setEnd
        }
        if (i == end) { //@cut
            res.push_back(end - start + 1); //@emit
            start = i + 1; //@next
        }
    }
    return res; //@ret
}
