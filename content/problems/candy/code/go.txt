func candy(ratings []int) int {
    n := len(ratings)
    candies := make([]int, n) //@init
    for i := range candies {
        candies[i] = 1 //@seed
    }
    for i := 1; i < n; i++ { //@leftPass
        if ratings[i] > ratings[i-1] { //@higherLeft
            candies[i] = candies[i-1] + 1 //@bumpLeft
        }
    }
    for i := n - 2; i >= 0; i-- { //@rightPass
        if ratings[i] > ratings[i+1] && candies[i] <= candies[i+1] { //@higherRight
            candies[i] = candies[i+1] + 1 //@bumpRight
        }
    }
    total := 0 //@sum
    for _, c := range candies { //@add
        total += c
    }
    return total //@ret
}
