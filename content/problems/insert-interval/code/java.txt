static int[][] insert(int[][] intervals, int[] newInterval) {
    List<int[]> res = new ArrayList<>(); //@init
    int i = 0, n = intervals.length; //@idx
    while (i < n && intervals[i][1] < newInterval[0]) { //@before
        res.add(intervals[i]); //@keep
        i++; //@advBefore
    }
    while (i < n && intervals[i][0] <= newInterval[1]) { //@overlap
        if (intervals[i][0] < newInterval[0]) { //@widenLo
            newInterval[0] = intervals[i][0];
        }
        if (intervals[i][1] > newInterval[1]) { //@widenHi
            newInterval[1] = intervals[i][1];
        }
        i++; //@advOverlap
    }
    res.add(newInterval); //@place
    while (i < n) { //@after
        res.add(intervals[i]); //@tail
        i++;
    }
    return res.toArray(new int[0][]); //@ret
}
