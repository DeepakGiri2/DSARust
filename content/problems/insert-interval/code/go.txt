func insert(intervals [][]int, newInterval []int) [][]int {
    res := [][]int{} //@init
    i, n := 0, len(intervals) //@idx
    for i < n && intervals[i][1] < newInterval[0] { //@before
        res = append(res, intervals[i]) //@keep
        i++ //@advBefore
    }
    for i < n && intervals[i][0] <= newInterval[1] { //@overlap
        if intervals[i][0] < newInterval[0] { //@widenLo
            newInterval[0] = intervals[i][0]
        }
        if intervals[i][1] > newInterval[1] { //@widenHi
            newInterval[1] = intervals[i][1]
        }
        i++ //@advOverlap
    }
    res = append(res, newInterval) //@place
    for i < n { //@after
        res = append(res, intervals[i]) //@tail
        i++
    }
    return res //@ret
}
