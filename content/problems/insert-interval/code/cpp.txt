vector<vector<int>> insert(vector<vector<int>>& intervals, vector<int> newInterval) {
    vector<vector<int>> res; //@init
    int i = 0, n = intervals.size(); //@idx
    while (i < n && intervals[i][1] < newInterval[0]) { //@before
        res.push_back(intervals[i]); //@keep
        i++; //@advBefore
    }
    while (i < n && intervals[i][0] <= newInterval[1]) { //@overlap
        if (intervals[i][0] < newInterval[0]) { //@widenLo
            newInterval[0] = intervals[i][0];
        }
        if (intervals[i][1] > newInterval[1]) { //@widenHi
            newInterval[1] = intervals[i][1];
        }
        i++; //@advOverlap
    }
    res.push_back(newInterval); //@place
    while (i < n) { //@after
        res.push_back(intervals[i]); //@tail
        i++;
    }
    return res; //@ret
}
