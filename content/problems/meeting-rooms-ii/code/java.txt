static int minMeetingRooms(int[][] intervals) {
    int n = intervals.length;
    int[] starts = new int[n], ends = new int[n]; //@split
    for (int i = 0; i < n; i++) {
        starts[i] = intervals[i][0];
        ends[i] = intervals[i][1];
    }
    Arrays.sort(starts);
    Arrays.sort(ends);
    int rooms = 0, best = 0; //@init
    int i = 0, j = 0; //@cursors
    while (i < n) { //@loop
        if (starts[i] < ends[j]) { //@overlap
            rooms++; //@open
            i++;
            if (rooms > best) { //@peak
                best = rooms; //@record
            }
        } else { //@free
            rooms--; //@close
            j++;
        }
    }
    return best; //@ret
}
