vector<vector<int>> generate(int numRows) {
    vector<vector<int>> res; //@init
    for (int i = 0; i < numRows; i++) { //@row
        vector<int> row(i + 1, 0); //@newrow
        row[0] = row[i] = 1; //@ends
        for (int j = 1; j < i; j++) { //@mid
            row[j] = res[i-1][j-1] + res[i-1][j]; //@sum
        }
        res.push_back(row); //@push
    }
    return res; //@ret
}
