#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> generate(int numRows) {
    vector<vector<int>> res;
    for (int i = 0; i < numRows; i++) {
        vector<int> row(i + 1, 0);
        row[0] = row[i] = 1;
        for (int j = 1; j < i; j++) {
            row[j] = res[i-1][j-1] + res[i-1][j];
        }
        res.push_back(row);
    }
    return res;
}

int main() {
    int numRows;
    cin >> numRows;
    auto res = generate(numRows);
    for (size_t i = 0; i < res.size(); i++) {
        for (size_t j = 0; j < res[i].size(); j++) {
            if (j) cout << " ";
            cout << res[i][j];
        }
        if (i + 1 < res.size()) cout << "\n";
    }
    cout << endl;
}
