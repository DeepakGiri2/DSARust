bool isValid(string s) {
    map<char, char> pairs = {{')','('}, {']','['}, {'}','{'}};
    vector<char> stack; //@init
    for (int i = 0; i < (int)s.size(); i++) { //@loop
        char c = s[i];
        if (c == '(' || c == '[' || c == '{') { //@isopen
            stack.push_back(c); //@push
        } else {
            if (stack.empty()) { //@empty
                return false; //@bad
            }
            char top = stack.back(); //@peek
            if (top != pairs[c]) { //@match
                return false; //@bad2
            }
            stack.pop_back(); //@pop
        }
    }
    return stack.empty(); //@end
}
