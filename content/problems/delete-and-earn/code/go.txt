func deleteAndEarn(nums []int) int {
    maxV := 0 //@scan
    for _, v := range nums {
        if v > maxV {
            maxV = v
        }
    }
    points := make([]int, maxV+1) //@bucket
    for _, v := range nums { //@fill
        points[v] += v //@add
    }
    prev2, prev1 := 0, 0 //@init
    for v := 1; v <= maxV; v++ { //@loop
        take := prev2 + points[v] //@take
        skip := prev1 //@skip
        best := take //@best
        if skip > best {
            best = skip
        }
        prev2, prev1 = prev1, best //@shift
    }
    return prev1 //@ret
}
