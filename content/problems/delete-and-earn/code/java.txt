static int deleteAndEarn(int[] nums) {
    int maxV = 0; //@scan
    for (int v : nums) {
        maxV = Math.max(maxV, v);
    }
    int[] points = new int[maxV + 1]; //@bucket
    for (int v : nums) { //@fill
        points[v] += v; //@add
    }
    int prev2 = 0, prev1 = 0; //@init
    for (int v = 1; v <= maxV; v++) { //@loop
        int take = prev2 + points[v]; //@take
        int skip = prev1; //@skip
        int best = Math.max(take, skip); //@best
        prev2 = prev1; prev1 = best; //@shift
    }
    return prev1; //@ret
}
