static int findMaximizedCapital(int k, int w, int[] profits, int[] capital) {
    int n = profits.length;
    Integer[] idx = new Integer[n]; //@sort
    for (int i = 0; i < n; i++) idx[i] = i;
    Arrays.sort(idx, (a, b) -> capital[a] - capital[b]);
    PriorityQueue<Integer> h = new PriorityQueue<>(Collections.reverseOrder()); //@init
    int i = 0; //@cursor
    for (int round = 0; round < k; round++) { //@loop
        while (i < n && capital[idx[i]] <= w) { //@afford
            h.add(profits[idx[i]]); //@offer
            i++;
        }
        if (h.isEmpty()) { //@none
            break;
        }
        w += h.poll(); //@take
    }
    return w; //@ret
}
