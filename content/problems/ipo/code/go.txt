func findMaximizedCapital(k int, w int, profits []int, capital []int) int {
    n := len(profits)
    idx := make([]int, n) //@sort
    for i := range idx {
        idx[i] = i
    }
    sort.Slice(idx, func(a, b int) bool { return capital[idx[a]] < capital[idx[b]] })
    h := &MaxHeap{} //@init
    i := 0 //@cursor
    for round := 0; round < k; round++ { //@loop
        for i < n && capital[idx[i]] <= w { //@afford
            h.Push(profits[idx[i]]) //@offer
            i++
        }
        if h.Len() == 0 { //@none
            break
        }
        w += h.Pop() //@take
    }
    return w //@ret
}
