func findPeakElement(nums []int) int {
    lo, hi := 0, len(nums)-1 //@bounds
    for lo < hi { //@loop
        mid := lo + (hi-lo)/2 //@mid
        if nums[mid] < nums[mid+1] { //@rising
            lo = mid + 1 //@goRight
        } else { //@falling
            hi = mid //@goLeft
        }
    }
    return lo //@ret
}
