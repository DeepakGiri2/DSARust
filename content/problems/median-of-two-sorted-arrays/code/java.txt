static double findMedianSortedArrays(int[] x, int[] y) {
    int[] a = x, b = y;
    if (a.length > b.length) { //@swap
        int[] t = a; a = b; b = t;
    }
    int m = a.length, n = b.length; //@sizes
    int half = (m + n + 1) / 2; //@half
    int lo = 0, hi = m; //@bounds
    while (lo <= hi) { //@loop
        int i = lo + (hi - lo) / 2; //@mid
        int j = half - i; //@partner
        long aLeft = i > 0 ? a[i - 1] : Long.MIN_VALUE, aRight = i < m ? a[i] : Long.MAX_VALUE; //@edges
        long bLeft = j > 0 ? b[j - 1] : Long.MIN_VALUE, bRight = j < n ? b[j] : Long.MAX_VALUE;
        if (aLeft <= bRight && bLeft <= aRight) { //@check
            if ((m + n) % 2 == 1) { //@odd
                return Math.max(aLeft, bLeft); //@retOdd
            }
            return (Math.max(aLeft, bLeft) + Math.min(aRight, bRight)) / 2.0; //@retEven
        }
        if (aLeft > bRight) { //@tooBig
            hi = i - 1; //@shrink
        } else { //@tooSmall
            lo = i + 1; //@grow
        }
    }
    return 0;
}
