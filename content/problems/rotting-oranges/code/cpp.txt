int orangesRotting(vector<vector<int>>& grid) {
    int rows = grid.size(), cols = grid[0].size();
    queue<pair<int,int>> q; //@init
    int fresh = 0;
    for (int r = 0; r < rows; r++) { //@scan
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == 2) q.push({r, c});
            else if (grid[r][c] == 1) fresh++;
        }
    }
    int minutes = 0; //@min0
    int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    while (!q.empty() && fresh > 0) { //@wave
        minutes++; //@tick
        int sz = q.size();
        for (int i = 0; i < sz; i++) { //@spread
            auto cell = q.front(); q.pop();
            for (int k = 0; k < 4; k++) {
                int nr = cell.first + dr[k], nc = cell.second + dc[k];
                if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] == 1) { //@check
                    grid[nr][nc] = 2; //@rot
                    fresh--;
                    q.push({nr, nc});
                }
            }
        }
    }
    if (fresh > 0) { //@chk
        return -1; //@bad
    }
    return minutes; //@ret
}
