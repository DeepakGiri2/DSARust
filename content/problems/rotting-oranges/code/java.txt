static int orangesRotting(int[][] grid) {
    int rows = grid.length, cols = grid[0].length;
    Queue<int[]> queue = new ArrayDeque<>(); //@init
    int fresh = 0;
    for (int r = 0; r < rows; r++) { //@scan
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == 2) queue.add(new int[]{r, c});
            else if (grid[r][c] == 1) fresh++;
        }
    }
    int minutes = 0; //@min0
    int[] dr = {1, -1, 0, 0}, dc = {0, 0, 1, -1};
    while (!queue.isEmpty() && fresh > 0) { //@wave
        minutes++; //@tick
        int sz = queue.size();
        for (int i = 0; i < sz; i++) { //@spread
            int[] cell = queue.poll();
            for (int k = 0; k < 4; k++) {
                int nr = cell[0] + dr[k], nc = cell[1] + dc[k];
                if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] == 1) { //@check
                    grid[nr][nc] = 2; //@rot
                    fresh--;
                    queue.add(new int[]{nr, nc});
                }
            }
        }
    }
    if (fresh > 0) { //@chk
        return -1; //@bad
    }
    return minutes; //@ret
}
