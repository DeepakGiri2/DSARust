bool repeatedSubstringPattern(string s) {
    int n = s.size();
    vector<int> lps(n, 0); //@lps
    for (int i = 1, length = 0; i < n; ) { //@build
        if (s[i] == s[length]) { //@match
            length++;
            lps[i] = length; //@set
            i++;
        } else if (length > 0) { //@fall
            length = lps[length - 1]; //@back
        } else {
            lps[i] = 0; //@zero
            i++;
        }
    }
    int longest = lps[n - 1]; //@longest
    int period = n - longest; //@period
    return longest > 0 && n % period == 0; //@ret
}
