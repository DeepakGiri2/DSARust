int pivotIndex(vector<int>& nums) {
    int total = 0; //@init
    for (int x : nums) {
        total += x; //@sum
    }
    int left = 0; //@left
    for (int i = 0; i < (int)nums.size(); i++) { //@loop
        int right = total - left - nums[i]; //@right
        if (left == right) { //@check
            return i; //@ret
        }
        left += nums[i]; //@advance
    }
    return -1; //@none
}
