bool equationsPossible(vector<string>& equations) {
    vector<int> parent(26); //@init
    for (int i = 0; i < 26; i++) {
        parent[i] = i; //@seed
    }
    for (auto& e : equations) { //@join
        if (e[1] == '=') { //@iseq
            int a = find(parent, e[0] - 'a'), b = find(parent, e[3] - 'a'); //@roots
            parent[a] = b; //@union
        }
    }
    for (auto& e : equations) { //@check
        if (e[1] == '!' && find(parent, e[0] - 'a') == find(parent, e[3] - 'a')) { //@clash
            return false; //@no
        }
    }
    return true; //@yes
}

int find(vector<int>& parent, int x) {
    while (parent[x] != x) { //@walk
        parent[x] = parent[parent[x]]; //@compress
        x = parent[x]; //@up
    }
    return x; //@root
}
