static boolean equationsPossible(String[] equations) {
    int[] parent = new int[26]; //@init
    for (int i = 0; i < 26; i++) {
        parent[i] = i; //@seed
    }
    for (String e : equations) { //@join
        if (e.charAt(1) == '=') { //@iseq
            int a = find(parent, e.charAt(0) - 'a'), b = find(parent, e.charAt(3) - 'a'); //@roots
            parent[a] = b; //@union
        }
    }
    for (String e : equations) { //@check
        if (e.charAt(1) == '!' && find(parent, e.charAt(0) - 'a') == find(parent, e.charAt(3) - 'a')) { //@clash
            return false; //@no
        }
    }
    return true; //@yes
}

static int find(int[] parent, int x) {
    while (parent[x] != x) { //@walk
        parent[x] = parent[parent[x]]; //@compress
        x = parent[x]; //@up
    }
    return x; //@root
}
