var values = []int{1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1} //@table
var symbols = []string{"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"}

func intToRoman(num int) string {
    out := "" //@init
    for i, v := range values { //@loop
        for num >= v { //@fits
            out += symbols[i] //@append
            num -= v //@subtract
        }
    }
    return out //@ret
}
