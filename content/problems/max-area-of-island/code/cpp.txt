int maxAreaOfIsland(vector<vector<char>>& grid) {
    int best = 0; //@init
    for (int r = 0; r < (int)grid.size(); r++) { //@rowloop
        for (int c = 0; c < (int)grid[0].size(); c++) { //@colloop
            if (grid[r][c] == '1') { //@found
                int area = dfs(grid, r, c); //@call
                if (area > best) { //@cmp
                    best = area; //@update
                }
            }
        }
    }
    return best; //@ret
}

int dfs(vector<vector<char>>& grid, int r, int c) {
    if (r < 0 || r >= (int)grid.size() || c < 0 || c >= (int)grid[0].size() || grid[r][c] != '1') { //@bounds
        return 0; //@retZero
    }
    grid[r][c] = '0'; //@sink
    return 1 + dfs(grid, r + 1, c) + dfs(grid, r - 1, c) + //@sum
        dfs(grid, r, c + 1) + dfs(grid, r, c - 1);
}
