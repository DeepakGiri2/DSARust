func maxAreaOfIsland(grid [][]byte) int {
    best := 0 //@init
    for r := range grid { //@rowloop
        for c := range grid[r] { //@colloop
            if grid[r][c] == '1' { //@found
                area := dfs(grid, r, c) //@call
                if area > best { //@cmp
                    best = area //@update
                }
            }
        }
    }
    return best //@ret
}

func dfs(grid [][]byte, r, c int) int {
    if r < 0 || r >= len(grid) || c < 0 || c >= len(grid[0]) || grid[r][c] != '1' { //@bounds
        return 0 //@retZero
    }
    grid[r][c] = '0' //@sink
    return 1 + dfs(grid, r+1, c) + dfs(grid, r-1, c) + //@sum
        dfs(grid, r, c+1) + dfs(grid, r, c-1)
}
