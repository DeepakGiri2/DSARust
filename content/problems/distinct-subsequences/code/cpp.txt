int numDistinct(string s, string t) {
    int m = s.size(), n = t.size();
    vector<vector<long long>> dp(m + 1, vector<long long>(n + 1, 0)); //@init
    for (int i = 0; i <= m; i++) {
        dp[i][0] = 1; //@base
    }
    for (int i = 1; i <= m; i++) { //@rowLoop
        for (int j = 1; j <= n; j++) { //@colLoop
            dp[i][j] = dp[i - 1][j]; //@skip
            if (s[i - 1] == t[j - 1]) { //@match
                dp[i][j] += dp[i - 1][j - 1]; //@use
            }
        }
    }
    return (int)dp[m][n]; //@ret
}
