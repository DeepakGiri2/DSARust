func numDistinct(s string, t string) int {
    m, n := len(s), len(t)
    dp := make([][]int, m+1) //@init
    for i := range dp {
        dp[i] = make([]int, n+1)
        dp[i][0] = 1 //@base
    }
    for i := 1; i <= m; i++ { //@rowLoop
        for j := 1; j <= n; j++ { //@colLoop
            dp[i][j] = dp[i-1][j] //@skip
            if s[i-1] == t[j-1] { //@match
                dp[i][j] += dp[i-1][j-1] //@use
            }
        }
    }
    return dp[m][n] //@ret
}
