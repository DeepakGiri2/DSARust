func twoSumII(numbers []int, target int) []int {
    l, r := 0, len(numbers)-1 //@init
    for l < r { //@loop
        sum := numbers[l] + numbers[r] //@sum
        if sum == target { //@cmp
            return []int{l + 1, r + 1} //@found
        } else if sum < target { //@less
            l++ //@moveL
        } else {
            r-- //@moveR
        }
    }
    return nil //@none
}
