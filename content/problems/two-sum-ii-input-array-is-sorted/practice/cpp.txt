#include <bits/stdc++.h>
using namespace std;

vector<int> twoSumII(vector<int>& numbers, int target) {
    int l = 0, r = (int)numbers.size() - 1;
    while (l < r) {
        int sum = numbers[l] + numbers[r];
        if (sum == target) {
            return {l + 1, r + 1};
        } else if (sum < target) {
            l++;
        } else {
            r--;
        }
    }
    return {};
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> numbers = readInts();
    int target = readInts()[0];
    vector<int> res = twoSumII(numbers, target);
    cout << res[0] << " " << res[1] << endl;
}
