const INF = 1 << 30

func networkDelayTime(times [][]int, n int, k int) int {
    adj := make([][][2]int, n+1) //@build
    for _, t := range times {
        adj[t[0]] = append(adj[t[0]], [2]int{t[1], t[2]})
    }
    dist := make([]int, n+1) //@init
    for i := range dist {
        dist[i] = INF
    }
    dist[k] = 0 //@seed
    visited := make([]bool, n+1) //@visited
    for iter := 0; iter < n; iter++ { //@loop
        u := -1 //@pick
        for v := 1; v <= n; v++ { //@scan
            if !visited[v] && dist[v] != INF && (u == -1 || dist[v] < dist[u]) {
                u = v //@closest
            }
        }
        if u == -1 { //@stuck
            break
        }
        visited[u] = true //@settle
        for _, e := range adj[u] { //@relax
            if dist[u]+e[1] < dist[e[0]] { //@shorter
                dist[e[0]] = dist[u] + e[1] //@update
            }
        }
    }
    ans := 0 //@max
    for v := 1; v <= n; v++ { //@check
        if dist[v] == INF { //@unreachable
            return -1 //@fail
        }
        if dist[v] > ans {
            ans = dist[v] //@bump
        }
    }
    return ans //@ret
}
