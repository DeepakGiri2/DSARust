static String shortestRoot(Node root, String word) {
    Node node = root; //@start
    for (int i = 0; i < word.length(); i++) { //@walk
        char c = word.charAt(i);
        if (!node.children.containsKey(c)) { //@absent
            break;
        }
        node = node.children.get(c); //@descend
        if (node.isWord) { //@isRoot
            return word.substring(0, i + 1); //@shortest
        }
    }
    return word; //@whole
}

static String replaceWords(List<String> dictionary, String sentence) {
    Node root = buildTrie(dictionary); //@build
    List<String> out = new ArrayList<>(); //@init
    for (String w : sentence.split(" ")) { //@loop
        out.add(shortestRoot(root, w)); //@replace
    }
    return String.join(" ", out); //@ret
}
