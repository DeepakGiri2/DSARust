func replaceWords(dictionary []string, sentence string) string {
    root := buildTrie(dictionary) //@build
    out := []string{} //@init
    for _, w := range strings.Fields(sentence) { //@loop
        out = append(out, shortestRoot(root, w)) //@replace
    }
    return strings.Join(out, " ") //@ret
}

func shortestRoot(root *Node, word string) string {
    node := root //@start
    for i := 0; i < len(word); i++ { //@walk
        c := word[i]
        if node.children[c] == nil { //@absent
            break
        }
        node = node.children[c] //@descend
        if node.isWord { //@isRoot
            return word[:i+1] //@shortest
        }
    }
    return word //@whole
}
