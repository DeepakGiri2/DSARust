static List<List<Integer>> res = new ArrayList<>(); //@init

static List<List<Integer>> subsetsWithDup(int[] nums) {
    Arrays.sort(nums); //@sort
    res.clear();
    back(nums, 0, new ArrayList<>()); //@call
    return res; //@ret
}

static void back(int[] nums, int start, List<Integer> cur) {
    res.add(new ArrayList<>(cur)); //@record
    for (int i = start; i < nums.length; i++) { //@loop
        if (i > start && nums[i] == nums[i - 1]) { //@dup
            continue; //@skip
        }
        cur.add(nums[i]);
        back(nums, i + 1, cur); //@rec
        cur.remove(cur.size() - 1);
    }
}
