int eraseOverlapIntervals(vector<vector<int>>& intervals) {
    sort(intervals.begin(), intervals.end(), [](auto& a, auto& b) { //@sort
        return a[1] < b[1];
    });
    int removed = 0; //@init
    int end = intervals[0][1]; //@first
    for (int i = 1; i < (int)intervals.size(); i++) { //@loop
        if (intervals[i][0] < end) { //@clash
            removed++; //@drop
        } else {
            end = intervals[i][1]; //@keep
        }
    }
    return removed; //@ret
}
