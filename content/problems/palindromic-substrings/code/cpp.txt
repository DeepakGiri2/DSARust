int countSubstrings(string s) {
    int count = 0; //@init
    for (int i = 0; i < (int)s.size(); i++) { //@loop
        count += expand(s, i, i); //@odd
        count += expand(s, i, i + 1); //@even
    }
    return count; //@ret
}

int expand(string& s, int l, int r) {
    int found = 0; //@expInit
    while (l >= 0 && r < (int)s.size() && s[l] == s[r]) { //@grow
        found++; //@hit
        l--; r++;
    }
    return found; //@expRet
}
