ListNode* deleteDuplicates(ListNode* head) {
    ListNode* cur = head; //@init
    while (cur != nullptr && cur->next != nullptr) { //@loop
        if (cur->val == cur->next->val) { //@same
            cur->next = cur->next->next; //@skip
        } else { //@diff
            cur = cur->next; //@advance
        }
    }
    return head; //@ret
}
