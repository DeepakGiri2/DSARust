const int INF = 1 << 30;

int minCostConnectPoints(vector<vector<int>>& points) {
    int n = points.size();
    vector<bool> inTree(n, false); //@init
    vector<int> best(n, INF); //@dist
    best[0] = 0; //@seed
    int total = 0; //@total
    for (int iter = 0; iter < n; iter++) { //@loop
        int u = -1; //@pick
        for (int v = 0; v < n; v++) { //@scan
            if (!inTree[v] && (u == -1 || best[v] < best[u])) {
                u = v; //@closest
            }
        }
        inTree[u] = true; //@add
        total += best[u]; //@pay
        for (int v = 0; v < n; v++) { //@relax
            if (!inTree[v]) {
                int d = abs(points[u][0] - points[v][0]) + abs(points[u][1] - points[v][1]); //@dist2
                if (d < best[v]) { //@cheaper
                    best[v] = d; //@update
                }
            }
        }
    }
    return total; //@ret
}
