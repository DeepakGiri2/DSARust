func maxProduct(nums []int) int {
    best, curMax, curMin := nums[0], nums[0], nums[0] //@init
    for i := 1; i < len(nums); i++ { //@loop
        v := nums[i]
        if v < 0 { //@negative
            curMax, curMin = curMin, curMax //@swap
        }
        curMax = max(v, curMax*v) //@setMax
        curMin = min(v, curMin*v) //@setMin
        if curMax > best { //@cmp
            best = curMax //@setBest
        }
    }
    return best //@ret
}
