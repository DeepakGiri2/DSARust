func removeStars(s string) string {
    stack := []byte{} //@init
    for i := 0; i < len(s); i++ { //@loop
        if s[i] == '*' { //@isStar
            stack = stack[:len(stack)-1] //@pop
        } else { //@isChar
            stack = append(stack, s[i]) //@push
        }
    }
    return string(stack) //@ret
}
