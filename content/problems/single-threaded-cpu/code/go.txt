func getOrder(tasks [][]int) []int {
    n := len(tasks)
    idx := make([]int, n) //@sort
    for i := range idx {
        idx[i] = i
    }
    sort.Slice(idx, func(a, b int) bool { return tasks[idx[a]][0] < tasks[idx[b]][0] })
    h := &MinHeap{} //@init
    time := 0 //@clock
    order := []int{} //@order
    i := 0 //@cursor
    for len(order) < n { //@loop
        for i < n && tasks[idx[i]][0] <= time { //@arrive
            h.Push([2]int{tasks[idx[i]][1], idx[i]}) //@enqueue
            i++
        }
        if h.Len() == 0 { //@idle
            time = tasks[idx[i]][0] //@jump
            continue
        }
        t := h.Pop() //@pick
        time += t[0] //@run
        order = append(order, t[1]) //@append
    }
    return order //@ret
}
