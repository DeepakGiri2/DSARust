func eliminateMaximum(dist []int, speed []int) int {
    n := len(dist)
    idx := make([]int, n) //@order
    for i := range idx {
        idx[i] = i
    }
    sort.Slice(idx, func(a, b int) bool { //@sort
        return dist[idx[a]]*speed[idx[b]] < dist[idx[b]]*speed[idx[a]]
    })
    for i := 0; i < n; i++ { //@loop
        m := idx[i] //@next
        if dist[m] <= i*speed[m] { //@late
            return i //@ret
        }
    }
    return n //@all
}
