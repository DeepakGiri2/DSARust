void fill(vector<vector<int>>& image, int r, int c, int from, int to) {
    if (r < 0 || r >= (int)image.size() || c < 0 || c >= (int)image[0].size() ||
        image[r][c] != from) { //@bounds
        return; //@stop
    }
    image[r][c] = to; //@paint
    fill(image, r + 1, c, from, to); //@down
    fill(image, r - 1, c, from, to); //@up
    fill(image, r, c + 1, from, to); //@right
    fill(image, r, c - 1, from, to); //@left
}

vector<vector<int>> floodFill(vector<vector<int>>& image, int sr, int sc, int color) {
    int start = image[sr][sc]; //@start
    if (start == color) { //@same
        return image;
    }
    fill(image, sr, sc, start, color); //@call
    return image; //@ret
}
