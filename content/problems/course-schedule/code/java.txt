static boolean canFinish(int numCourses, int[][] prerequisites) {
    List<List<Integer>> adj = new ArrayList<>(); //@init
    for (int i = 0; i < numCourses; i++) adj.add(new ArrayList<>());
    for (int[] p : prerequisites) { //@build
        adj.get(p[0]).add(p[1]);
    }
    int[] state = new int[numCourses]; //@state
    for (int i = 0; i < numCourses; i++) { //@mainloop
        if (!dfs(i, adj, state)) { //@call
            return false; //@retfalse
        }
    }
    return true; //@rettrue
}

static boolean dfs(int course, List<List<Integer>> adj, int[] state) {
    if (state[course] == 1) { //@cycle
        return false; //@backedge
    }
    if (state[course] == 2) { //@donecheck
        return true; //@skip
    }
    state[course] = 1; //@visiting
    for (int dep : adj.get(course)) { //@deploop
        if (!dfs(dep, adj, state)) { //@rec
            return false; //@propagate
        }
    }
    state[course] = 2; //@done
    return true; //@ok
}
