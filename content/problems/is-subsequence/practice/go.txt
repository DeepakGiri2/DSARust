package main

import (
    "bufio"
    "fmt"
    "os"
    "strings"
)

func isSubsequence(s string, t string) bool {
    i, j := 0, 0
    for i < len(s) && j < len(t) {
        if s[i] == t[j] {
            i++
        }
        j++
    }
    return i == len(s)
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    s, _ := reader.ReadString('\n')
    t, _ := reader.ReadString('\n')
    s = strings.TrimRight(s, "\r\n")
    t = strings.TrimRight(t, "\r\n")
    fmt.Println(isSubsequence(s, t))
}
