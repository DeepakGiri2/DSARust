bool hasCycle(ListNode* head) {
    ListNode* slow = head; //@init
    ListNode* fast = head; //@init2
    while (fast != nullptr && fast->next != nullptr) { //@loop
        slow = slow->next; //@slow
        fast = fast->next->next; //@fast
        if (slow == fast) { //@check
            return true; //@true
        }
    }
    return false; //@false
}
