#include <bits/stdc++.h>
using namespace std;

int removeDuplicates(vector<int>& nums) {
    if (nums.empty()) {
        return 0;
    }
    int slow = 0;
    for (int fast = 1; fast < (int)nums.size(); fast++) {
        if (nums[fast] != nums[slow]) {
            slow++;
            nums[slow] = nums[fast];
        }
    }
    return slow + 1;
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> nums = readInts();
    int k = removeDuplicates(nums);
    cout << k << "\n";
    for (int i = 0; i < k; i++) {
        if (i) cout << " ";
        cout << nums[i];
    }
    cout << "\n";
}
