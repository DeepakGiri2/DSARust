static int[][] kClosest(int[][] points, int k) {
    PriorityQueue<int[]> h = new PriorityQueue<>((a, b) -> //@init
        (b[0] * b[0] + b[1] * b[1]) - (a[0] * a[0] + a[1] * a[1]));
    for (int[] p : points) { //@loop
        int d = p[0] * p[0] + p[1] * p[1]; //@dist
        h.add(p); //@push
        if (h.size() > k) { //@check
            h.poll(); //@pop
        }
    }
    return h.toArray(new int[0][]); //@ret
}
