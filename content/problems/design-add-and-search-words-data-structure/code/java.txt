class WordDictionary {
    Map<Character, WordDictionary> children = new LinkedHashMap<>(); //@init
    boolean isWord = false; //@flag

    void addWord(String word) {
        WordDictionary node = this; //@addStart
        for (int i = 0; i < word.length(); i++) { //@addLoop
            char c = word.charAt(i);
            if (!node.children.containsKey(c)) { //@missing
                node.children.put(c, new WordDictionary()); //@create
            }
            node = node.children.get(c); //@descend
        }
        node.isWord = true; //@mark
    }

    boolean search(String word) {
        return match(word, 0); //@searchStart
    }

    boolean match(String word, int i) {
        if (i == word.length()) { //@end
            return isWord; //@atWord
        }
        char c = word.charAt(i); //@char
        if (c == '.') { //@wildcard
            for (WordDictionary child : children.values()) { //@tryAll
                if (child.match(word, i + 1)) { //@branch
                    return true;
                }
            }
            return false; //@noBranch
        }
        if (!children.containsKey(c)) { //@exact
            return false; //@dead
        }
        return children.get(c).match(word, i + 1); //@recurse
    }
}
