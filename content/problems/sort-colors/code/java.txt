static void sortColors(int[] nums) {
    int low = 0, mid = 0, high = nums.length - 1; //@init
    while (mid <= high) { //@loop
        if (nums[mid] == 0) { //@is0
            int tmp = nums[low]; nums[low] = nums[mid]; nums[mid] = tmp; //@swaplo
            low++; //@inclow
            mid++; //@incmid
        } else if (nums[mid] == 1) { //@is1
            mid++; //@justmid
        } else {
            int tmp = nums[mid]; nums[mid] = nums[high]; nums[high] = tmp; //@swaphi
            high--; //@dechigh
        }
    }
}
