func sortColors(nums []int) {
    low, mid, high := 0, 0, len(nums)-1 //@init
    for mid <= high { //@loop
        if nums[mid] == 0 { //@is0
            nums[low], nums[mid] = nums[mid], nums[low] //@swaplo
            low++ //@inclow
            mid++ //@incmid
        } else if nums[mid] == 1 { //@is1
            mid++ //@justmid
        } else {
            nums[mid], nums[high] = nums[high], nums[mid] //@swaphi
            high-- //@dechigh
        }
    }
}
