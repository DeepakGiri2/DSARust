void sortColors(vector<int>& nums) {
    int low = 0, mid = 0, high = (int)nums.size() - 1; //@init
    while (mid <= high) { //@loop
        if (nums[mid] == 0) { //@is0
            swap(nums[low], nums[mid]); //@swaplo
            low++; //@inclow
            mid++; //@incmid
        } else if (nums[mid] == 1) { //@is1
            mid++; //@justmid
        } else {
            swap(nums[mid], nums[high]); //@swaphi
            high--; //@dechigh
        }
    }
}
