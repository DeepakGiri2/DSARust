int uniquePathsWithObstacles(vector<vector<int>>& grid) {
    int rows = grid.size(), cols = grid[0].size();
    vector<vector<int>> dp(rows, vector<int>(cols, 0)); //@init
    for (int r = 0; r < rows; r++) { //@rowLoop
        for (int c = 0; c < cols; c++) { //@colLoop
            if (grid[r][c] == 1) { //@blocked
                dp[r][c] = 0; //@zero
                continue;
            }
            if (r == 0 && c == 0) { //@start
                dp[r][c] = 1; //@seed
                continue;
            }
            int total = 0; //@sum
            if (r > 0) { //@fromAbove
                total += dp[r - 1][c]; //@addAbove
            }
            if (c > 0) { //@fromLeft
                total += dp[r][c - 1]; //@addLeft
            }
            dp[r][c] = total; //@store
        }
    }
    return dp[rows - 1][cols - 1]; //@ret
}
