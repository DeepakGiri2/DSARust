func uniquePathsWithObstacles(grid [][]int) int {
    rows, cols := len(grid), len(grid[0])
    dp := make([][]int, rows) //@init
    for r := range dp {
        dp[r] = make([]int, cols)
    }
    for r := 0; r < rows; r++ { //@rowLoop
        for c := 0; c < cols; c++ { //@colLoop
            if grid[r][c] == 1 { //@blocked
                dp[r][c] = 0 //@zero
                continue
            }
            if r == 0 && c == 0 { //@start
                dp[r][c] = 1 //@seed
                continue
            }
            total := 0 //@sum
            if r > 0 { //@fromAbove
                total += dp[r-1][c] //@addAbove
            }
            if c > 0 { //@fromLeft
                total += dp[r][c-1] //@addLeft
            }
            dp[r][c] = total //@store
        }
    }
    return dp[rows-1][cols-1] //@ret
}
