double maxProbability(int n, vector<vector<int>>& edges, vector<double>& succProb,
                      int start, int end) {
    vector<vector<pair<int, double>>> adj(n); //@build
    for (size_t i = 0; i < edges.size(); i++) {
        adj[edges[i][0]].push_back({edges[i][1], succProb[i]});
        adj[edges[i][1]].push_back({edges[i][0], succProb[i]});
    }
    vector<double> prob(n, 0.0); //@init
    prob[start] = 1.0; //@seed
    vector<bool> visited(n, false); //@visited
    for (int iter = 0; iter < n; iter++) { //@loop
        int u = -1; //@pick
        for (int v = 0; v < n; v++) { //@scan
            if (!visited[v] && (u == -1 || prob[v] > prob[u])) {
                u = v; //@highest
            }
        }
        if (prob[u] == 0.0) { //@stuck
            break;
        }
        visited[u] = true; //@settle
        for (auto& e : adj[u]) { //@relax
            double p = prob[u] * e.second; //@multiply
            if (p > prob[e.first]) { //@better
                prob[e.first] = p; //@update
            }
        }
    }
    return prob[end]; //@ret
}
