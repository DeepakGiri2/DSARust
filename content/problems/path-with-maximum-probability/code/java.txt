static double maxProbability(int n, int[][] edges, double[] succProb, int start, int end) {
    List<double[]>[] adj = new List[n]; //@build
    for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
    for (int i = 0; i < edges.length; i++) {
        adj[edges[i][0]].add(new double[]{edges[i][1], succProb[i]});
        adj[edges[i][1]].add(new double[]{edges[i][0], succProb[i]});
    }
    double[] prob = new double[n]; //@init
    prob[start] = 1.0; //@seed
    boolean[] visited = new boolean[n]; //@visited
    for (int iter = 0; iter < n; iter++) { //@loop
        int u = -1; //@pick
        for (int v = 0; v < n; v++) { //@scan
            if (!visited[v] && (u == -1 || prob[v] > prob[u])) {
                u = v; //@highest
            }
        }
        if (prob[u] == 0.0) { //@stuck
            break;
        }
        visited[u] = true; //@settle
        for (double[] e : adj[u]) { //@relax
            double p = prob[u] * e[1]; //@multiply
            if (p > prob[(int) e[0]]) { //@better
                prob[(int) e[0]] = p; //@update
            }
        }
    }
    return prob[end]; //@ret
}
