bool carPooling(vector<vector<int>>& trips, int capacity) {
    sort(trips.begin(), trips.end(), [](auto& a, auto& b) { return a[1] < b[1]; }); //@sort
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> h; //@init
    int onboard = 0; //@onboard
    for (auto& t : trips) { //@loop
        while (!h.empty() && h.top().first <= t[1]) { //@dropoff
            onboard -= h.top().second; h.pop(); //@leave
        }
        onboard += t[0]; //@board
        if (onboard > capacity) { //@over
            return false; //@no
        }
        h.push({t[2], t[0]}); //@push
    }
    return true; //@yes
}
