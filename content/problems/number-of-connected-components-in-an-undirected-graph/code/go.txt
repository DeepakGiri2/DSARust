func countComponents(n int, edges [][]int) int {
    parent := make([]int, n) //@init
    for i := range parent { //@seed
        parent[i] = i //@self
    }
    count := n //@count
    for _, e := range edges { //@loop
        a, b := find(parent, e[0]), find(parent, e[1]) //@find
        if a != b { //@check
            parent[a] = b //@union
            count-- //@merge
        }
    }
    return count //@ret
}

func find(parent []int, x int) int {
    for parent[x] != x {
        parent[x] = parent[parent[x]]
        x = parent[x]
    }
    return x
}
