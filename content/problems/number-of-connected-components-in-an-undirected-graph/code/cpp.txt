int countComponents(int n, vector<vector<int>>& edges) {
    vector<int> parent(n); //@init
    for (int i = 0; i < n; i++) { //@seed
        parent[i] = i; //@self
    }
    int count = n; //@count
    for (auto& e : edges) { //@loop
        int a = find(parent, e[0]), b = find(parent, e[1]); //@find
        if (a != b) { //@check
            parent[a] = b; //@union
            count--; //@merge
        }
    }
    return count; //@ret
}
