void nextPermutation(vector<int>& nums) {
    int n = nums.size();
    int i = n - 2; //@start
    while (i >= 0 && nums[i] >= nums[i + 1]) { //@descend
        i--; //@back
    }
    if (i >= 0) { //@pivot
        int j = n - 1; //@scan
        while (nums[j] <= nums[i]) { //@bigger
            j--; //@downj
        }
        swap(nums[i], nums[j]); //@swap
    }
    for (int lo = i + 1, hi = n - 1; lo < hi; lo++, hi--) { //@flip
        swap(nums[lo], nums[hi]); //@mirror
    }
}
