int kthSmallest(TreeNode* root, int k) {
    stack<TreeNode*> stk; //@init
    TreeNode* cur = root; //@setcur
    while (cur != nullptr || !stk.empty()) { //@loop
        while (cur != nullptr) { //@dive
            stk.push(cur); //@push
            cur = cur->left; //@left
        }
        cur = stk.top(); stk.pop(); //@pop
        k--; //@dec
        if (k == 0) { //@check
            return cur->val; //@found
        }
        cur = cur->right; //@right
    }
    return -1; //@notfound
}
