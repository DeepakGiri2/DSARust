static final int INF = 1 << 30;

static int swimInWater(int[][] grid) {
    int n = grid.length;
    int[][] best = new int[n][n]; //@init
    boolean[][] visited = new boolean[n][n]; //@visited
    for (int[] row : best) Arrays.fill(row, INF);
    best[0][0] = grid[0][0]; //@seed
    int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
    for (int iter = 0; iter < n * n; iter++) { //@loop
        int br = -1, bc = -1; //@pick
        for (int r = 0; r < n; r++) { //@scan
            for (int c = 0; c < n; c++) {
                if (!visited[r][c] && best[r][c] != INF && (br < 0 || best[r][c] < best[br][bc])) {
                    br = r; bc = c; //@lowest
                }
            }
        }
        if (br < 0) { //@stuck
            break;
        }
        if (br == n - 1 && bc == n - 1) { //@arrived
            return best[br][bc]; //@ret
        }
        visited[br][bc] = true; //@settle
        for (int[] d : dirs) { //@relax
            int nr = br + d[0], nc = bc + d[1];
            if (nr >= 0 && nr < n && nc >= 0 && nc < n && !visited[nr][nc]) {
                int t = Math.max(best[br][bc], grid[nr][nc]); //@level
                if (t < best[nr][nc]) { //@better
                    best[nr][nc] = t; //@update
                }
            }
        }
    }
    return -1; //@none
}
