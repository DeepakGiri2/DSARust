func strStr(haystack string, needle string) int {
    m := len(needle)
    lps := make([]int, m) //@lps
    for i, length := 1, 0; i < m; { //@build
        if needle[i] == needle[length] { //@match
            length++
            lps[i] = length //@set
            i++
        } else if length > 0 { //@fall
            length = lps[length-1] //@back
        } else {
            lps[i] = 0 //@zero
            i++
        }
    }

    for i, j := 0, 0; i < len(haystack); { //@scan
        if haystack[i] == needle[j] { //@cmp
            i++
            j++ //@adv
            if j == m { //@done
                return i - j //@found
            }
        } else if j > 0 { //@shift
            j = lps[j-1] //@jump
        } else {
            i++ //@next
        }
    }
    return -1 //@none
}
