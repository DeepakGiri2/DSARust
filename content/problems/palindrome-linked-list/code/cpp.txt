bool isPalindrome(ListNode* head) {
    ListNode *slow = head, *fast = head; //@mid
    while (fast != nullptr && fast->next != nullptr) { //@midLoop
        slow = slow->next;
        fast = fast->next->next;
    }
    ListNode* prev = nullptr; //@revInit
    while (slow != nullptr) { //@revLoop
        ListNode* next = slow->next;
        slow->next = prev; //@flip
        prev = slow;
        slow = next;
    }
    ListNode *left = head, *right = prev; //@cmpInit
    while (right != nullptr) { //@cmpLoop
        if (left->val != right->val) { //@cmp
            return false; //@no
        }
        left = left->next; //@advance
        right = right->next;
    }
    return true; //@yes
}
