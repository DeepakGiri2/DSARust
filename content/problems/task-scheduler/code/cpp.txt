int leastInterval(vector<char>& tasks, int n) {
    unordered_map<char, int> count; //@count
    int maxCount = 0; //@maxInit
    for (char t : tasks) { //@tally
        count[t]++; //@inc
        if (count[t] > maxCount) { //@cmpMax
            maxCount = count[t]; //@setMax
        }
    }
    int ties = 0; //@ties
    for (auto& kv : count) { //@tieLoop
        if (kv.second == maxCount) { //@isTie
            ties++; //@addTie
        }
    }
    int slots = (maxCount - 1) * (n + 1) + ties; //@formula
    if (slots < (int)tasks.size()) { //@compare
        return tasks.size(); //@noIdle
    }
    return slots; //@ret
}
