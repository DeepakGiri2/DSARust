static List<Integer> rightSideView(TreeNode root) {
    List<Integer> res = new ArrayList<>(); //@init
    if (root == null) { //@empty
        return res; //@retempty
    }
    Queue<TreeNode> q = new ArrayDeque<>(); //@enq
    q.add(root);
    while (!q.isEmpty()) { //@while
        int size = q.size(); //@size
        for (int i = 0; i < size; i++) { //@front
            TreeNode node = q.poll(); //@pop
            if (i == size - 1) { //@islast
                res.add(node.val); //@take
            }
            if (node.left != null) { //@left
                q.add(node.left);
            }
            if (node.right != null) { //@right
                q.add(node.right);
            }
        }
    }
    return res; //@ret
}
