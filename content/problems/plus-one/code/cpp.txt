vector<int> plusOne(vector<int>& digits) {
    for (int i = digits.size() - 1; i >= 0; i--) { //@loop
        if (digits[i] < 9) { //@check
            digits[i]++; //@inc
            return digits; //@done
        }
        digits[i] = 0; //@zero
    }
    digits.insert(digits.begin(), 1); //@grow
    return digits;
}
