static int[] plusOne(int[] digits) {
    for (int i = digits.length - 1; i >= 0; i--) { //@loop
        if (digits[i] < 9) { //@check
            digits[i]++; //@inc
            return digits; //@done
        }
        digits[i] = 0; //@zero
    }
    int[] out = new int[digits.length + 1]; //@grow
    out[0] = 1;
    return out;
}
