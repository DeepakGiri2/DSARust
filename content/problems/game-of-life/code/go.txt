func gameOfLife(board [][]int) {
    m, n := len(board), len(board[0])
    for r := 0; r < m; r++ { //@row
        for c := 0; c < n; c++ { //@col
            live := neighbours(board, r, c) //@count
            if board[r][c]&1 == 1 && (live == 2 || live == 3) { //@survive
                board[r][c] |= 2 //@keep
            } else if board[r][c]&1 == 0 && live == 3 { //@born
                board[r][c] |= 2 //@spawn
            }
        }
    }
    for r := 0; r < m; r++ { //@second
        for c := 0; c < n; c++ {
            board[r][c] >>= 1 //@shift
        }
    }
}

func neighbours(board [][]int, r, c int) int {
    live := 0 //@zero
    for dr := -1; dr <= 1; dr++ { //@scan
        for dc := -1; dc <= 1; dc++ {
            if dr == 0 && dc == 0 {
                continue
            }
            nr, nc := r+dr, c+dc
            if nr >= 0 && nr < len(board) && nc >= 0 && nc < len(board[0]) && board[nr][nc]&1 == 1 { //@alive
                live++ //@bump
            }
        }
    }
    return live //@total
}
