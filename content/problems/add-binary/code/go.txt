func addBinary(a string, b string) string {
    i, j := len(a)-1, len(b)-1 //@init
    carry := 0 //@carry
    out := []byte{} //@out
    for i >= 0 || j >= 0 || carry > 0 { //@loop
        sum := carry //@start
        if i >= 0 { //@takeA
            sum += int(a[i] - '0')
            i--
        }
        if j >= 0 { //@takeB
            sum += int(b[j] - '0')
            j--
        }
        out = append(out, byte('0'+sum%2)) //@bit
        carry = sum / 2 //@nextCarry
    }
    for l, r := 0, len(out)-1; l < r; l, r = l+1, r-1 { //@reverse
        out[l], out[r] = out[r], out[l]
    }
    return string(out) //@ret
}
