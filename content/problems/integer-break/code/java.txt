static int integerBreak(int n) {
    int[] dp = new int[n + 1]; //@init
    dp[1] = 1; //@seed
    for (int i = 2; i <= n; i++) { //@loop
        for (int j = 1; j < i; j++) { //@split
            int best = j * (i - j); //@plain
            if (j * dp[i - j] > best) { //@deeper
                best = j * dp[i - j]; //@useDp
            }
            if (best > dp[i]) { //@improve
                dp[i] = best; //@store
            }
        }
    }
    return dp[n]; //@ret
}
