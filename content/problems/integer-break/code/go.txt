func integerBreak(n int) int {
    dp := make([]int, n+1) //@init
    dp[1] = 1 //@seed
    for i := 2; i <= n; i++ { //@loop
        for j := 1; j < i; j++ { //@split
            best := j * (i - j) //@plain
            if j*dp[i-j] > best { //@deeper
                best = j * dp[i-j] //@useDp
            }
            if best > dp[i] { //@improve
                dp[i] = best //@store
            }
        }
    }
    return dp[n] //@ret
}
