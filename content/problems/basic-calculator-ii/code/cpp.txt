int calculate(string s) {
    vector<int> stack; //@init
    int num = 0;
    char op = '+'; //@state
    for (int i = 0; i < (int)s.size(); i++) {
        char c = s[i]; //@ch
        if (isdigit(c)) { //@digit
            num = num * 10 + (c - '0'); //@build
        }
        if ((c != ' ' && !isdigit(c)) || i == (int)s.size() - 1) { //@flush
            switch (op) { //@apply
                case '+': stack.push_back(num); break; //@plus
                case '-': stack.push_back(-num); break; //@minus
                case '*': stack.back() *= num; break; //@times
                case '/': stack.back() /= num; break; //@divide
            }
            op = c; //@remember
            num = 0;
        }
    }
    int sum = 0; //@sum
    for (int v : stack) { //@add
        sum += v;
    }
    return sum; //@ret
}
