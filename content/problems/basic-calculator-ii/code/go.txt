func calculate(s string) int {
    stack := []int{} //@init
    num := 0
    op := byte('+') //@state
    for i := 0; i < len(s); i++ {
        c := s[i] //@ch
        if c >= '0' && c <= '9' { //@digit
            num = num*10 + int(c-'0') //@build
        }
        if (c != ' ' && (c < '0' || c > '9')) || i == len(s)-1 { //@flush
            switch op { //@apply
            case '+':
                stack = append(stack, num) //@plus
            case '-':
                stack = append(stack, -num) //@minus
            case '*':
                stack[len(stack)-1] *= num //@times
            case '/':
                stack[len(stack)-1] /= num //@divide
            }
            op = c //@remember
            num = 0
        }
    }
    sum := 0 //@sum
    for _, v := range stack { //@add
        sum += v
    }
    return sum //@ret
}
