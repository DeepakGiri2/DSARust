static int[] parent;

static int find(int x) {
    while (parent[x] != x) { //@walk
        parent[x] = parent[parent[x]]; //@compress
        x = parent[x]; //@up
    }
    return x; //@root
}

static int[] findRedundantConnection(int[][] edges) {
    parent = new int[edges.length + 1]; //@init
    for (int i = 0; i < parent.length; i++) {
        parent[i] = i; //@seed
    }
    for (int[] e : edges) { //@loop
        int a = find(e[0]), b = find(e[1]); //@find
        if (a == b) { //@cycle
            return e; //@ret
        }
        parent[a] = b; //@union
    }
    return new int[0]; //@none
}
