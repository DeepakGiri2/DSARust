vector<int> parent;

int find(int x) {
    while (parent[x] != x) { //@walk
        parent[x] = parent[parent[x]]; //@compress
        x = parent[x]; //@up
    }
    return x; //@root
}

vector<int> findRedundantConnection(vector<vector<int>>& edges) {
    parent.assign(edges.size() + 1, 0); //@init
    for (int i = 0; i < (int)parent.size(); i++) {
        parent[i] = i; //@seed
    }
    for (auto& e : edges) { //@loop
        int a = find(e[0]), b = find(e[1]); //@find
        if (a == b) { //@cycle
            return e; //@ret
        }
        parent[a] = b; //@union
    }
    return {}; //@none
}
