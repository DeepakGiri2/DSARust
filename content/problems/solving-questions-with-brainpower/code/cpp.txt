long long mostPoints(vector<vector<int>>& questions) {
    int n = questions.size();
    vector<long long> dp(n + 1, 0); //@init
    for (int i = n - 1; i >= 0; i--) { //@loop
        int points = questions[i][0], brain = questions[i][1]; //@read
        int next = i + brain + 1; //@jump
        long long take = points; //@take
        if (next <= n) { //@inRange
            take += dp[next]; //@addFuture
        }
        long long skip = dp[i + 1]; //@skip
        if (take > skip) { //@compare
            dp[i] = take; //@storeTake
        } else {
            dp[i] = skip; //@storeSkip
        }
    }
    return dp[0]; //@ret
}
