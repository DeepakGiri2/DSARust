static int firstMissingPositive(int[] nums) {
    int n = nums.length; //@n
    for (int i = 0; i < n; i++) { //@loop
        while (nums[i] > 0 && nums[i] <= n && nums[nums[i] - 1] != nums[i]) { //@can
            int j = nums[i] - 1; //@home
            int t = nums[i]; nums[i] = nums[j]; nums[j] = t; //@swap
        }
    }
    for (int i = 0; i < n; i++) { //@check
        if (nums[i] != i + 1) { //@gap
            return i + 1; //@ret
        }
    }
    return n + 1; //@all
}
