package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
    "strings"
)

type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

func isBalanced(root *TreeNode) bool {
    return dfs(root) != -1
}

func dfs(root *TreeNode) int {
    if root == nil {
        return 0
    }
    left := dfs(root.Left)
    right := dfs(root.Right)
    if left == -1 || right == -1 {
        return -1
    }
    if left-right > 1 || right-left > 1 {
        return -1
    }
    return 1 + max(left, right)
}

func buildTree(tokens []string) *TreeNode {
    if len(tokens) == 0 || tokens[0] == "n" {
        return nil
    }
    v, _ := strconv.Atoi(tokens[0])
    root := &TreeNode{Val: v}
    queue := []*TreeNode{root}
    i := 1
    for len(queue) > 0 && i < len(tokens) {
        cur := queue[0]
        queue = queue[1:]
        if i < len(tokens) {
            if tokens[i] != "n" {
                v, _ := strconv.Atoi(tokens[i])
                cur.Left = &TreeNode{Val: v}
                queue = append(queue, cur.Left)
            }
            i++
        }
        if i < len(tokens) {
            if tokens[i] != "n" {
                v, _ := strconv.Atoi(tokens[i])
                cur.Right = &TreeNode{Val: v}
                queue = append(queue, cur.Right)
            }
            i++
        }
    }
    return root
}

func main() {
    reader := bufio.NewReader(os.Stdin)
    line, _ := reader.ReadString('\n')
    root := buildTree(strings.Fields(line))
    fmt.Println(isBalanced(root))
}
