vector<int> productExceptSelf(vector<int>& nums) {
    int n = (int)nums.size(); //@n
    vector<int> res(n, 1); //@res
    int prefix = 1; //@pre
    for (int i = 0; i < n; i++) { //@fwd
        res[i] = prefix; //@setpre
        prefix *= nums[i]; //@mulpre
    }
    int postfix = 1; //@post
    for (int i = n - 1; i >= 0; i--) { //@bwd
        res[i] *= postfix; //@setpost
        postfix *= nums[i]; //@mulpost
    }
    return res; //@return
}
