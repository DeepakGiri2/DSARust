int numSquares(int n) {
    vector<int> dp(n + 1, 0); //@init
    for (int i = 1; i <= n; i++) { //@loop
        dp[i] = i; //@worst
        for (int k = 1; k * k <= i; k++) { //@inner
            if (dp[i - k * k] + 1 < dp[i]) { //@better
                dp[i] = dp[i - k * k] + 1; //@update
            }
        }
    }
    return dp[n]; //@ret
}
