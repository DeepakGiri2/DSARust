#include <bits/stdc++.h>
using namespace std;

vector<int> replaceElements(vector<int>& arr) {
    int mx = -1;
    for (int i = (int)arr.size() - 1; i >= 0; i--) {
        int cur = arr[i];
        arr[i] = mx;
        if (cur > mx) {
            mx = cur;
        }
    }
    return arr;
}

vector<int> readInts() {
    string line;
    getline(cin, line);
    istringstream ss(line);
    vector<int> nums;
    int x;
    while (ss >> x) nums.push_back(x);
    return nums;
}

int main() {
    vector<int> arr = readInts();
    vector<int> res = replaceElements(arr);
    for (size_t i = 0; i < res.size(); i++) {
        if (i) cout << " ";
        cout << res[i];
    }
    cout << endl;
}
