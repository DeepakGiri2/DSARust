vector<int> replaceElements(vector<int>& arr) {
    int mx = -1; //@init
    for (int i = (int)arr.size() - 1; i >= 0; i--) { //@loop
        int cur = arr[i]; //@read
        arr[i] = mx; //@replace
        if (cur > mx) { //@check
            mx = cur; //@update
        }
    }
    return arr; //@ret
}
