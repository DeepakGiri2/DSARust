static int minDistance(String word1, String word2) {
    int m = word1.length(), n = word2.length();
    int[][] dp = new int[m + 1][n + 1]; //@init
    for (int i = 0; i <= m; i++) {
        dp[i][0] = i; //@baseRow
    }
    for (int j = 0; j <= n; j++) { //@baseCol
        dp[0][j] = j;
    }
    for (int i = 1; i <= m; i++) { //@rowLoop
        for (int j = 1; j <= n; j++) { //@colLoop
            if (word1.charAt(i - 1) == word2.charAt(j - 1)) { //@same
                dp[i][j] = dp[i - 1][j - 1]; //@free
            } else { //@differ
                int best = dp[i - 1][j - 1]; //@replace
                if (dp[i - 1][j] < best) { //@delete
                    best = dp[i - 1][j];
                }
                if (dp[i][j - 1] < best) { //@insert
                    best = dp[i][j - 1];
                }
                dp[i][j] = best + 1; //@store
            }
        }
    }
    return dp[m][n]; //@ret
}
