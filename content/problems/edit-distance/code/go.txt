func minDistance(word1 string, word2 string) int {
    m, n := len(word1), len(word2)
    dp := make([][]int, m+1) //@init
    for i := range dp {
        dp[i] = make([]int, n+1)
        dp[i][0] = i //@baseRow
    }
    for j := 0; j <= n; j++ { //@baseCol
        dp[0][j] = j
    }
    for i := 1; i <= m; i++ { //@rowLoop
        for j := 1; j <= n; j++ { //@colLoop
            if word1[i-1] == word2[j-1] { //@same
                dp[i][j] = dp[i-1][j-1] //@free
            } else { //@differ
                best := dp[i-1][j-1] //@replace
                if dp[i-1][j] < best { //@delete
                    best = dp[i-1][j]
                }
                if dp[i][j-1] < best { //@insert
                    best = dp[i][j-1]
                }
                dp[i][j] = best + 1 //@store
            }
        }
    }
    return dp[m][n] //@ret
}
