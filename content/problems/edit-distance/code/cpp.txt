int minDistance(string word1, string word2) {
    int m = word1.size(), n = word2.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0)); //@init
    for (int i = 0; i <= m; i++) {
        dp[i][0] = i; //@baseRow
    }
    for (int j = 0; j <= n; j++) { //@baseCol
        dp[0][j] = j;
    }
    for (int i = 1; i <= m; i++) { //@rowLoop
        for (int j = 1; j <= n; j++) { //@colLoop
            if (word1[i - 1] == word2[j - 1]) { //@same
                dp[i][j] = dp[i - 1][j - 1]; //@free
            } else { //@differ
                int best = dp[i - 1][j - 1]; //@replace
                if (dp[i - 1][j] < best) { //@delete
                    best = dp[i - 1][j];
                }
                if (dp[i][j - 1] < best) { //@insert
                    best = dp[i][j - 1];
                }
                dp[i][j] = best + 1; //@store
            }
        }
    }
    return dp[m][n]; //@ret
}
